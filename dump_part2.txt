    def _load_assets(self) -> None:
        if self._registry is None:
            self._palette.set_assets([])
            return
        try:
            self._palette.set_assets(self._registry.all())
        except Exception:
            self._palette.set_assets([])

    def _get_drop_position(self) -> tuple:
        """РџРѕР·РёС†РёСЏ РґР»СЏ РЅРѕРІРѕРіРѕ РєРѕРјРїРѕРЅРµРЅС‚Р° вЂ” РїРѕРґ РјС‹С€СЊСЋ РёР»Рё РІ С†РµРЅС‚СЂРµ РІРёРґР°."""
        try:
            from PySide6.QtGui import QCursor
            global_pos = QCursor.pos()
            local_pos = self._canvas.viewport().mapFromGlobal(global_pos)
            if self._canvas.viewport().rect().contains(local_pos):
                scene_pos = self._canvas.mapToScene(local_pos)
                return (scene_pos.x(), scene_pos.y())
            center = self._canvas.viewport().rect().center()
            scene_pos = self._canvas.mapToScene(center)
            return (scene_pos.x(), scene_pos.y())
        except Exception:
            return (0.0, 0.0)

    def _on_add_asset(self, asset_id: str) -> None:
        """Р”РІРѕР№РЅРѕР№ РєР»РёРє РІ РїР°Р»РёС‚СЂРµ вЂ” РґРѕР±Р°РІРёС‚СЊ РєРѕРјРїРѕРЅРµРЅС‚ РЅР° СЃС†РµРЅСѓ."""
        # Р•СЃР»Рё РІ preview вЂ” РІС‹Р№С‚Рё
        if self._preview_mode:
            self._exit_preview()

        if self._registry is None:
            self.statusBar().showMessage("Р РµРµСЃС‚СЂ Р°СЃСЃРµС‚РѕРІ РЅРµРґРѕСЃС‚СѓРїРµРЅ", 3000)
            return

        asset = self._registry.get(asset_id)
        if asset is None:
            self.statusBar().showMessage(
                f"РђСЃСЃРµС‚ {asset_id} РЅРµ РЅР°Р№РґРµРЅ", 3000)
            return

        # РџРѕР·РёС†РёСЏ вЂ” РїРѕРґ РєСѓСЂСЃРѕСЂРѕРј РјС‹С€Рё РёР»Рё РІ С†РµРЅС‚СЂРµ РІРёРґР°
        drop_x, drop_y = self._get_drop_position()

        comp = Component(
            asset_id=asset_id,
            x=drop_x,
            y=drop_y,
            rotation=0.0,
            scale=1.0,
            name=asset.name,
        )

        # Р§РµСЂРµР· undo-РєРѕРјР°РЅРґСѓ
        cmd = AddComponentCommand(
            composite=self._current_composite,
            comp=comp,
            on_create_item=self._create_component_item,
            on_remove_item=self._remove_component_item,
        )
        self._undo_stack.push(cmd)

        self.statusBar().showMessage(
            f"Р”РѕР±Р°РІР»РµРЅ: {asset.name} (id={comp.id[:8]}). "
            f"Р’СЃРµРіРѕ: {self._current_composite.component_count()}",
            4000,
        )

    def _reset_scene(self) -> None:
        """РћС‡РёСЃС‚РёС‚СЊ СЃС†РµРЅСѓ Рё РјРѕРґРµР»СЊ."""
        for item in list(self._scene.items()):
            self._scene.removeItem(item)
        self._items_by_comp_id.clear()
        self._current_composite = Asset(name="РќРѕРІС‹Р№ Р·Р°РјРѕРє", type_="tower_body")
        self._current_asset_id = None
        self._current_asset_name = "РќРѕРІС‹Р№ Р·Р°РјРѕРє"
        self._nav_stack.clear()
        self._properties.clear()
        self._update_title()

    def _update_title(self) -> None:
        crumbs = [name for (_, name) in self._nav_stack]
        crumbs.append(self._current_asset_name)
        path = " вЂє ".join(crumbs)
        self.setWindowTitle(f"Constructor вЂ” {path}")
        if hasattr(self, "_act_back"):
            self._act_back.setEnabled(bool(self._nav_stack))

    def _on_new(self) -> None:
        self._preview_mode = False
        self._edit_backup = None
        self._reset_scene()
        self.statusBar().showMessage("РќРѕРІС‹Р№ Р·Р°РјРѕРє", 3000)

    def _on_save(self) -> None:
        if self._current_asset_id is None:
            self._on_save_as()
            return

        # РћР±РЅРѕРІР»СЏРµРј РёРјСЏ Рё СЃРѕС…СЂР°РЅСЏРµРј
        self._current_composite.name = self._current_asset_name

        # V13: СЃРѕС…СЂР°РЅРёС‚СЊ ground line РёР· СЃС†РµРЅС‹
        if self._scene is not None and hasattr(
            self._scene, "ground_line_y"
        ):
            self._current_composite.ground_line_y = (
                self._scene.ground_line_y()
            )
            self._current_composite.ground_line_visible = (
                self._scene.ground_line_visible()
            )

        try:
            save_asset(self._current_composite)
        except StorageError as e:
            QMessageBox.critical(
                self, "РћС€РёР±РєР° СЃРѕС…СЂР°РЅРµРЅРёСЏ",
                f"РќРµ СѓРґР°Р»РѕСЃСЊ СЃРѕС…СЂР°РЅРёС‚СЊ:\n\n{e}",
            )
            return

        # РћР±РЅРѕРІРёС‚СЊ СЂРµРµСЃС‚СЂ Рё РїР°Р»РёС‚СЂСѓ вЂ” РЅРѕРІС‹Р№/РёР·РјРµРЅС‘РЅРЅС‹Р№ Р°СЃСЃРµС‚
        if self._registry is not None:
            self._registry.load_all()
        self._load_assets()

        self.statusBar().showMessage(
            f"РЎРѕС…СЂР°РЅРµРЅРѕ: {self._current_asset_name}", 3000,
        )

    def _on_save_as(self) -> None:
        name, ok = QInputDialog.getText(
            self, "РЎРѕС…СЂР°РЅРёС‚СЊ РєР°Рє",
            "РРјСЏ РєРѕРјРїРѕР·РёС‚РЅРѕРіРѕ Р°СЃСЃРµС‚Р°:",
            text=self._current_asset_name,
        )
        if not ok:
            return
        name = name.strip()
        if not name:
            return

        # РќРѕРІС‹Р№ Asset вЂ” СЃ РЅРѕРІС‹Рј uuid, РєРѕРїРёСЂСѓРµРј СЃРѕРґРµСЂР¶РёРјРѕРµ
        new_asset = Asset(name=name, type_="tower_body")
        new_asset.components = dict(self._current_composite.components)

        # V13: СЃРєРѕРїРёСЂРѕРІР°С‚СЊ ground line
        if self._scene is not None and hasattr(
            self._scene, "ground_line_y"
        ):
            new_asset.ground_line_y = self._scene.ground_line_y()
            new_asset.ground_line_visible = (
                self._scene.ground_line_visible()
            )

        try:
            save_asset(new_asset)
        except StorageError as e:
            QMessageBox.critical(
                self, "РћС€РёР±РєР° СЃРѕС…СЂР°РЅРµРЅРёСЏ",
                f"РќРµ СѓРґР°Р»РѕСЃСЊ СЃРѕС…СЂР°РЅРёС‚СЊ:\n\n{e}",
            )
            return

        self._current_composite = new_asset
        self._current_asset_id = new_asset.id
        self._current_asset_name = name
        self._update_title()

        # РћР±РЅРѕРІРёС‚СЊ СЂРµРµСЃС‚СЂ Рё РїР°Р»РёС‚СЂСѓ вЂ” РЅРѕРІС‹Р№ Р°СЃСЃРµС‚
        if self._registry is not None:
            self._registry.load_all()
        self._load_assets()

        self.statusBar().showMessage(
            f"РЎРѕР·РґР°РЅ: {name} (id={new_asset.id[:8]})", 3000,
        )

    def _on_open(self) -> None:
        if self._preview_mode:
            self._preview_mode = False
            self._edit_backup = None

        # Р—Р°РіСЂСѓР¶Р°РµРј РІСЃРµ Р°СЃСЃРµС‚С‹, С„РёР»СЊС‚СЂСѓРµРј РєРѕРјРїРѕР·РёС‚РЅС‹Рµ
        try:
            all_assets = list_assets()
        except Exception as e:
            QMessageBox.warning(
                self, "РћС€РёР±РєР°", f"РќРµ СѓРґР°Р»РѕСЃСЊ Р·Р°РіСЂСѓР·РёС‚СЊ:\n\n{e}")
            return

        composites = [a for a in all_assets if a.components]
        if not composites:
            QMessageBox.information(
                self, "РќРµС‚ РєРѕРјРїРѕР·РёС‚РЅС‹С… Р°СЃСЃРµС‚РѕРІ",
                "РЎРЅР°С‡Р°Р»Р° СЃРѕР·РґР°Р№ Рё СЃРѕС…СЂР°РЅРё Р·Р°РјРѕРє С‡РµСЂРµР· "
                "В«РЎРѕС…СЂР°РЅРёС‚СЊ РєР°РєВ».",
            )
            return

        dlg = AssetOpenDialog(composites, mode="open", parent=self)
        if dlg.exec() != AssetOpenDialog.DialogCode.Accepted:
            # РњРѕР¶РµС‚ Р±С‹С‚СЊ СѓРґР°Р»РµРЅРёРµ
            del_id = dlg.deleted_id()
            if del_id:
                self._do_delete_asset(del_id)
            return

        # РњРѕР¶РµС‚ Р±С‹С‚СЊ РЅР°Р¶Р°Р»Рё В«РЈРґР°Р»РёС‚СЊВ»
        del_id = dlg.deleted_id()
        if del_id:
            self._do_delete_asset(del_id)
            return

        aid = dlg.selected_id()
        if not aid:
            return

        # РїСЂР°РІРєРё РёР· Vector Editor (РіСЂСѓРїРїС‹, anchors, РєРѕРЅС‚СѓСЂС‹)
        if self._registry is not None:

        self._load_composite(target)

    def _load_composite(self, asset: Asset) -> None:
        """Р—Р°РіСЂСѓР·РёС‚СЊ РєРѕРјРїРѕР·РёС‚РЅС‹Р№ Asset РІ СЃС†РµРЅСѓ."""
        self._reset_scene()

        self._current_composite = asset
        self._current_asset_id = asset.id
        self._current_asset_name = asset.name
        self._update_title()

        # N РѕС‚РґРµР»СЊРЅС‹С… items вЂ” РїРѕ РѕРґРЅРѕРјСѓ РЅР° РєРѕРјРїРѕРЅРµРЅС‚ РІРµСЂС…РЅРµРіРѕ СѓСЂРѕРІРЅСЏ.
        # Р’Р»РѕР¶РµРЅРЅС‹Рµ composite РІРЅСѓС‚СЂРё РєР°Р¶РґРѕРіРѕ item РѕС‚СЂРёСЃСѓСЋС‚СЃСЏ СЂРµРєСѓСЂСЃРёРІРЅРѕ.
        n_orphan = 0
        for comp in asset.components.values():
            ref_asset = None
            if self._registry is not None:
                ref_asset = self._registry.get(comp.asset_id)
            if ref_asset is None:
                n_orphan += 1

            item = ComponentItem(
                comp, asset=ref_asset, registry=self._registry,
            )
            item.moved.connect(self._on_item_moved)
            item.enter_requested.connect(self._on_enter_composite)
            item.drag_finished.connect(self._on_item_drag_finished)
            item.param_changed.connect(self._on_prop_param_override)
            self._items_by_comp_id[comp.id] = item
            self._scene.addItem(item)

        msg = (
            f"Р—Р°РіСЂСѓР¶РµРЅРѕ: {asset.name} В· "
            f"{len(asset.components)} РєРѕРјРїРѕРЅРµРЅС‚РѕРІ"
        )
        if n_orphan:
            msg += f" В· {n_orphan} Р±РёС‚С‹С…"
        self.statusBar().showMessage(msg, 5000)

        # V13: Р·Р°РіСЂСѓР·РёС‚СЊ РѕРїРѕСЂРЅСѓСЋ Р»РёРЅРёСЋ РёР· РєРѕРјРїРѕР·РёС‚Р°
        if self._scene is not None:
            g_y = getattr(asset, "ground_line_y", None)
            g_v = bool(getattr(asset, "ground_line_visible", True))
            self._scene.set_ground_line(g_y, visible=g_v)
            if g_y is not None:
                self._ground_spin.blockSignals(True)
                self._ground_spin.setValue(float(g_y))
                self._ground_spin.blockSignals(False)
                self._act_ground.blockSignals(True)
                self._act_ground.setChecked(g_v)
                self._act_ground.blockSignals(False)
            else:
                self._act_ground.blockSignals(True)
                self._act_ground.setChecked(False)
                self._act_ground.blockSignals(False)

        # РџСЂРёРјРµРЅРёС‚СЊ РїСЂР°РІРёР»Р° СЃСЂР°Р·Сѓ РїСЂРё Р·Р°РіСЂСѓР·РєРµ
        self._apply_visibility_rules()
        self._apply_slot_visibility()

        # V-A: Р°РІС‚Рѕ-zoom РїРѕРґ РІСЃРµ РєРѕРјРїРѕРЅРµРЅС‚С‹ РєРѕРјРїРѕР·РёС‚Р°.
        from PySide6.QtCore import QTimer as _QT
        _QT.singleShot(0, self._fit_to_scene)

    def _fit_to_scene(self) -> None:
        """V-A: РїРѕРґРѕРіРЅР°С‚СЊ РІРёРґ canvas РїРѕРґ bbox РІСЃРµС… РєРѕРјРїРѕРЅРµРЅС‚РѕРІ."""
        if self._scene is None:
            return
        r = self._scene.itemsBoundingRect()
        if r.isEmpty():
            return
        # РњРёРЅРёРјСѓРј 1Г—1 Рј вЂ” С‡С‚РѕР±С‹ РЅРµ Р·СѓРјРёС‚СЊСЃСЏ РґРѕ Р±РµСЃРєРѕРЅРµС‡РЅРѕСЃС‚Рё
        # РЅР° РїРѕС‡С‚Рё РїСѓСЃС‚РѕР№ СЃС†РµРЅРµ.
        w = max(r.width(), 1.0)
        h = max(r.height(), 1.0)
        cx = r.center().x()
        cy = r.center().y()
        x0 = cx - w / 2
        x1 = cx + w / 2
        y0 = cy - h / 2
        y1 = cy + h / 2
        self._canvas.fit_to_rect(x0, y0, x1, y1)

    def _on_enter_composite(self, asset_id: str) -> None:
        """Р”РІРѕР№РЅРѕР№ РєР»РёРє РїРѕ composite вЂ” РІРѕР№С‚Рё РІРЅСѓС‚СЂСЊ."""
        if self._current_asset_id is None:

                "Р§С‚РѕР±С‹ РІРѕР№С‚Рё РІРЅСѓС‚СЂСЊ, РЅР°РґРѕ СЃРѕС…СЂР°РЅРёС‚СЊ С‚РµРєСѓС‰РёР№ composite.\n\n"

                | QMessageBox.StandardButton.No,

        if sub is None:
            QMessageBox.warning(

    def _apply_property(self, comp_id: str) -> None:
        """РџРѕСЃР»Рµ undo/redo СЃРІРѕР№СЃС‚РІР° вЂ” РѕР±РЅРѕРІРёС‚СЊ item + РїР°РЅРµР»СЊ."""
        comp = self._current_composite.get_component(comp_id)
        item = self._items_by_comp_id.get(comp_id)
        if item is not None:
            item.apply_from_component()

        if (self._properties._current_comp is not None
                and self._properties._current_comp.id == comp_id):
            self._properties.update_values(
                comp.x, comp.y, comp.rotation, comp.scale,
            )
            if hasattr(self._properties, "_layer_spin"):
                self._properties._muted = True
                self._properties._layer_spin.setValue(
                    int(getattr(comp, "layer", 0))
                )
                self._properties._muted = False

    def _on_prop_value_changed(
        self, comp_id: str, field: str, value: float,
    ) -> None:
        """РџР°РЅРµР»СЊ РёР·РјРµРЅРёР»Р° СЃРІРѕР№СЃС‚РІРѕ вЂ” С‡РµСЂРµР· undo-РєРѕРјР°РЅРґСѓ."""
        comp = self._current_composite.get_component(comp_id)
        item = self._items_by_comp_id.get(comp_id)
        if comp is None or item is None:
            return

        if field not in ("x", "y", "rotation", "scale", "layer"):
            return

        old = getattr(comp, field, None)
        if old is None:
            return

        new = int(value) if field == "layer" else float(value)

        if abs(float(old) - float(new)) < 1e-9:
            return

        cmd = SetPropertyCommand(
            comp, field, old, new,
            on_apply=lambda: self._apply_property(comp_id),
        )
        self._undo_stack.push(cmd)

    def _on_prop_layer_shift(
        self, comp_id: str, direction: int,
    ) -> None:
        """РЎРґРІРёРЅСѓС‚СЊ СЃР»РѕР№ РЅР° +1 / -1 вЂ” С‡РµСЂРµР· undo-РєРѕРјР°РЅРґСѓ."""
        comp = self._current_composite.get_component(comp_id)
        if comp is None:
            return

        old = int(getattr(comp, "layer", 0))
        new = old + int(direction)

        cmd = SetPropertyCommand(
            comp, "layer", old, new,
            on_apply=lambda: self._apply_property(comp_id),
        )
        self._undo_stack.push(cmd)

    def _on_prop_filled_changed(

        item.update()

    def _on_prop_fill_pattern_changed(
        self, comp_id: str, pattern: str,
    ) -> None:
        """V11: СЃРјРµРЅР° С‚РµРєСЃС‚СѓСЂС‹ Р·Р°Р»РёРІРєРё С‡РµСЂРµР· undo-РєРѕРјР°РЅРґСѓ."""

        new = str(pattern or "")
        if old == new:
            return

        def _apply():
            it = self._items_by_comp_id.get(comp_id)
            if it is not None:
                it.update()

        cmd = SetPropertyCommand(
            comp, "fill_pattern", old, new,

    def _on_prop_locked_changed(
        self, comp_id: str, locked: bool,
    ) -> None:
        """V12: РїРѕР»СЊР·РѕРІР°С‚РµР»СЊ РїРµСЂРµРєР»СЋС‡РёР» Р±Р»РѕРєРёСЂРѕРІРєСѓ."""

    def _on_prop_param_override(
        self, comp_id: str, name: str, value: float,
    ) -> None:
        """РџРѕР»СЊР·РѕРІР°С‚РµР»СЊ РјРµРЅСЏРµС‚ РїР°СЂР°РјРµС‚СЂ sub-Р°СЃСЃРµС‚Р° вЂ” С‡РµСЂРµР· РєРѕРјР°РЅРґСѓ."""
        comp = self._current_composite.get_component(comp_id)
        if comp is None:

    def _reflow_children(
        self, parent_id: str, visited: set | None = None,
    ) -> None:
        """Р РµРєСѓСЂСЃРёРІРЅРѕ РїРµСЂРµСЃС‡РёС‚Р°С‚СЊ РїРѕР·РёС†РёРё РїРѕС‚РѕРјРєРѕРІ."""
        if self._current_composite is None:
            return
        if visited is None:
            visited = set()
        if parent_id in visited:
            return
        visited.add(parent_id)

        parent_item = self._items_by_comp_id.get(parent_id)
        if parent_item is None:
            return

        from PySide6.QtCore import QPointF

        for child_id, child_item in self._items_by_comp_id.items():
            if child_id in visited:
                continue
            child_comp = self._current_composite.get_component(child_id)
            if child_comp is None:
                continue
            if child_comp.attach_to != parent_id:
                continue
            if not child_comp.attach_anchor or not child_comp.parent_anchor:
                continue

            # V9d-4: РµСЃР»Рё parent_anchor вЂ” СЃР»РѕС‚ (slot_*),
            # Р±РµСЂС‘Рј РёР· slots_local РІРјРµСЃС‚Рѕ anchors_local.
            if child_comp.parent_anchor.startswith("slot_"):
                parent_anchors = parent_item.slots_local()
            else:
                parent_anchors = parent_item.anchors_local()
            child_anchors = child_item.anchors_local()

            p_local = parent_anchors.get(child_comp.parent_anchor)
            c_local = child_anchors.get(child_comp.attach_anchor)
            if p_local is None or c_local is None:
                continue

            parent_scene = parent_item.mapToScene(
                QPointF(p_local[0], p_local[1])
            )

            new_x = parent_scene.x() - c_local[0]
            new_y = parent_scene.y() - c_local[1]

            child_comp.x = new_x
            child_comp.y = new_y
            child_item.setPos(new_x, new_y)
            child_item.update()

            self._reflow_children(child_id, visited)

    def _apply_visibility_rules(self) -> None:
        """РџСЂРёРјРµРЅРёС‚СЊ РІСЃРµ visibility-РїСЂР°РІРёР»Р° Рє items.

        Р”Р»СЏ РєР°Р¶РґРѕРіРѕ РїСЂР°РІРёР»Р°:
          1. РќР°С…РѕРґРёРј РєРѕРјРїРѕРЅРµРЅС‚-С‚СЂРёРіРіРµСЂ.
          2. Р‘РµСЂС‘Рј Р·РЅР°С‡РµРЅРёРµ РїР°СЂР°РјРµС‚СЂР° (override РёР»Рё СЂРѕРґРЅРѕРµ).
          3. РћС†РµРЅРёРІР°РµРј СѓСЃР»РѕРІРёРµ.
          4. Р”Р»СЏ РІСЃРµС… РєРѕРјРїРѕРЅРµРЅС‚РѕРІ РїСЂРёРјРµРЅСЏРµРј action_for в†’ show/hide.
        """
        if self._current_composite is None:
            return
        rules = self._current_composite.visibility_rules.values()
        if not rules:
            return

        for rule in rules:
            if not rule.enabled:
                continue

            comp = self._current_composite.get_component(
                rule.component_id,
            )
            if comp is None:
                continue

            # Р—РЅР°С‡РµРЅРёРµ РїР°СЂР°РјРµС‚СЂР°
            ov = comp.param_overrides.get(rule.parameter_name)
            if ov is not None:
                value = ov
            else:
                ref = None
                if self._registry is not None:
                    ref = self._registry.get(comp.asset_id)
                if ref is None:
                    continue
                p = None
                for pp in ref.parameters_list():
                    if pp.name == rule.parameter_name:
                        p = pp
                        break
                if p is None:
                    continue
                value = p.value

            condition = rule.evaluate(value)

            # РџСЂРёРјРµРЅСЏРµРј РєРѕ РІСЃРµРј Р·Р°РіСЂСѓР¶РµРЅРЅС‹Рј РєРѕРјРїРѕРЅРµРЅС‚Р°Рј
            for cid, item in self._items_by_comp_id.items():
                # V11: РїСЂРѕРїСѓСЃРєР°РµРј РєРѕРјРїРѕРЅРµРЅС‚С‹ РЅР° СЃР»РѕС‚Р°С… вЂ”
                # РёРјРё СѓРїСЂР°РІР»СЏРµС‚ _apply_slot_visibility, РЅРµ rules.
                c2 = self._current_composite.get_component(cid)
                if (
                    c2 is not None
                    and c2.parent_anchor.startswith("slot_")
                ):
                    continue
                action = rule.action_for(cid, condition)
                if action == "show" and not item.isVisible():
                    item.setVisible(True)
                elif action == "hide" and item.isVisible():
                    item.setVisible(False)

    def _apply_slot_visibility(self) -> None:
        """V11: РїРѕРєР°Р·Р°С‚СЊ/СЃРєСЂС‹С‚СЊ РєРѕРјРїРѕРЅРµРЅС‚С‹ РЅР° СЃР»РѕС‚Р°С… РїРѕ РІС‹СЃРѕС‚Рµ СЂРѕРґРёС‚РµР»СЏ.

        Р›РѕРіРёРєР°:
          visible = wall_height >= slot_y_from_bottom + clearance
        РіРґРµ wall_height Рё slot_y СѓР¶Рµ СѓС‡РёС‚С‹РІР°СЋС‚ param_overrides.

        Р•СЃР»Рё slot_policy.enabled == False (РёР»Рё policy=None) вЂ” РїСЂРѕРїСѓСЃРє.
        РљРѕРјРїРѕРЅРµРЅС‚ РќР• СѓРґР°Р»СЏРµС‚СЃСЏ вЂ” С‚РѕР»СЊРєРѕ setVisible(False/True).
        """
        if self._current_composite is None:
            return

        for cid, item in list(self._items_by_comp_id.items()):
            comp = self._current_composite.get_component(cid)
            if comp is None:
                continue
            # РўРѕР»СЊРєРѕ РєРѕРјРїРѕРЅРµРЅС‚С‹, РїСЂРёРІСЏР·Р°РЅРЅС‹Рµ Рє СЃР»РѕС‚Р°Рј
            if not comp.parent_anchor.startswith("slot_"):
                continue
            if not comp.attach_to:
                continue

            parent_item = self._items_by_comp_id.get(comp.attach_to)
            if parent_item is None:
                continue
            parent_comp = self._current_composite.get_component(
                comp.attach_to,
            )
            if parent_comp is None:
                continue

            policy = getattr(parent_comp, "slot_policy", None)
            if policy is None:
                continue
            if not policy.get("enabled", False):
                continue
            clearance = float(policy.get("clearance", 1.5))

            wall_h = parent_item.wall_height_m()
            if wall_h <= 0:
                continue

            slot_h = parent_item.slot_y_from_bottom(
                comp.parent_anchor,
            )
            if slot_h is None:
                if item.isVisible():
                    item.setVisible(False)
                continue

            window_h = item.wall_height_m()
            visible = wall_h >= slot_h + window_h + clearance
            if item.isVisible() != visible:
                item.setVisible(visible)

        """РЎРѕС…СЂР°РЅРёС‚СЊ С‚РµРєСѓС‰РёР№ РєРѕРјРїРѕР·РёС‚ Рё РѕС‡РёСЃС‚РёС‚СЊ СЃС†РµРЅСѓ."""
        if not self._preview_mode:
            self._edit_backup = (
                self._current_composite,
                self._current_asset_id,
                self._current_asset_name,
            )
            self._preview_mode = True

        self._reset_scene()
        self._preview_mode = True

        self.setWindowTitle("Constructor вЂ” РџСЂРѕСЃРјРѕС‚СЂ")
        if hasattr(self, "_properties"):
            self._properties.setEnabled(False)
            self._properties.setVisible(False)

        self.statusBar().showMessage(
            "Р РµР¶РёРј РїСЂРѕСЃРјРѕС‚СЂР°. РљР»РёРє РїРѕ Р°СЃСЃРµС‚Сѓ вЂ” РїРѕРєР°Р·Р°С‚СЊ.", 5000,
        )

    def _show_asset_preview(self, asset_id: str) -> None:
        """РџРѕРєР°Р·Р°С‚СЊ Р°СЃСЃРµС‚ РІ СЂРµР¶РёРјРµ РїСЂРѕСЃРјРѕС‚СЂР° (СЃРѕ СЃР»РѕСЏРјРё Рё Р·Р°Р»РёРІРєРѕР№)."""
        if self._registry is None:
            return
        asset = self._registry.get(asset_id)
        if asset is None:
            return

        for item in list(self._scene.items()):
            self._scene.removeItem(item)
        self._items_by_comp_id.clear()

        # Р•СЃР»Рё РєРѕРјРїРѕР·РёС‚ вЂ” РѕС‚РґРµР»СЊРЅС‹Рµ items РЅР° РєР°Р¶РґС‹Р№ sub-РєРѕРјРїРѕРЅРµРЅС‚,
        # РєР°Рє РІ РѕР±С‹С‡РЅРѕРј СЂРµРґР°РєС‚РѕСЂРµ (Р·Р°Р»РёРІРєР° Рё СЃР»РѕРё СЂР°Р±РѕС‚Р°СЋС‚)
        if asset.components:
            for comp in asset.components.values():
                ref = self._registry.get(comp.asset_id)
                item = ComponentItem(
                    comp, asset=ref, registry=self._registry,
                )
                item.setFlag(
                    QGraphicsItem.GraphicsItemFlag.ItemIsMovable, False,
                )
                item.setFlag(
                    QGraphicsItem.GraphicsItemFlag.ItemIsSelectable, False,
                )
                self._scene.addItem(item)
        else:
            # РџСЂРѕСЃС‚РѕР№ Р°СЃСЃРµС‚
            comp = Component(
                asset_id=asset.id,
                x=0.0, y=0.0,
                rotation=0.0, scale=1.0,
                name=asset.name,
            )
            item = ComponentItem(
                comp, asset=asset, registry=self._registry,
            )
            item.setFlag(
                QGraphicsItem.GraphicsItemFlag.ItemIsMovable, False,
            )
            item.setFlag(
                QGraphicsItem.GraphicsItemFlag.ItemIsSelectable, False,
            )
            self._scene.addItem(item)

        self.setWindowTitle(f"Constructor вЂ” РџСЂРѕСЃРјРѕС‚СЂ: {asset.name}")
        self.statusBar().showMessage(f"РџСЂРѕСЃРјРѕС‚СЂ: {asset.name}", 3000)
    def _on_asset_selected(self, asset_id: str) -> None:
        """РћРґРёРЅРѕС‡РЅС‹Р№ РєР»РёРє: РІ СЂРµР¶РёРјРµ РїСЂРѕСЃРјРѕС‚СЂР° вЂ” РїРѕРєР°Р·Р°С‚СЊ Р°СЃСЃРµС‚."""
        if self._preview_mode:
            self._show_asset_preview(asset_id)
