### mount_role.py ###
"""
MountRole - С„РёРєСЃРёСЂРѕРІР°РЅРЅС‹Рµ СЂРѕР»Рё С‚РѕС‡РµРє РєСЂРµРїР»РµРЅРёСЏ.

Р§С‚Рѕ РјРѕР¶РЅРѕ РїРѕСЃС‚Р°РІРёС‚СЊ РІ СЌС‚Сѓ С‚РѕС‡РєСѓ.
None РґРѕРїСѓСЃС‚РёРј РґР»СЏ legacy-Р°СЃСЃРµС‚РѕРІ РїРѕСЃР»Рµ РјРёРіСЂР°С†РёРё
(СЃРёСЃС‚РµРјР° РЅРµ СѓРіР°РґС‹РІР°РµС‚ СЂРѕР»СЊ Р°РІС‚РѕРјР°С‚РёС‡РµСЃРєРё).
"""

from __future__ import annotations

from enum import StrEnum


class MountRole(StrEnum):
    """Р РѕР»СЊ С‚РѕС‡РєРё РєСЂРµРїР»РµРЅРёСЏ."""
    WINDOW = "window"
    DOOR = "door"
    FOUNDATION = "foundation"
    CORNICE = "cornice"
    DECOR = "decor"


def parse_role(value) -> MountRole | None:
    """Р Р°Р·РѕР±СЂР°С‚СЊ СЂРѕР»СЊ РёР· СЃС‚СЂРѕРєРё. None РґР»СЏ РїСѓСЃС‚С‹С…/РЅРµРІР°Р»РёРґРЅС‹С…."""
    if value is None or value == "":
        return None
    if isinstance(value, MountRole):
        return value
    try:
        return MountRole(str(value))
    except (ValueError, TypeError):
        return None


def role_to_str(role) -> str | None:
    """РЎРµСЂРёР°Р»РёР·РѕРІР°С‚СЊ СЂРѕР»СЊ РІ СЃС‚СЂРѕРєСѓ."""
    if role is None:
        return None
    if isinstance(role, MountRole):
        return role.value
    return str(role)


### mount_point.py ###
"""
MountPoint + Distribution + MountLocation.

MountPoint - РѕРїСЂРµРґРµР»РµРЅРёРµ С‚РѕС‡РєРё РєСЂРµРїР»РµРЅРёСЏ РІ Р›РћРљРђР›Р¬РќР«РҐ РєРѕРѕСЂРґРёРЅР°С‚Р°С… Asset.
Distribution - РїСЂР°РІРёР»Рѕ СЂР°Р·РјРЅРѕР¶РµРЅРёСЏ (count_x/y, spacing_x/y).
MountLocation - РІС‹С‡РёСЃР»РµРЅРЅР°СЏ РїРѕР·РёС†РёСЏ (РїСЂРѕРёР·РІРѕРґРЅР°СЏ, РќР• СЃРѕС…СЂР°РЅСЏРµС‚СЃСЏ).
"""

from __future__ import annotations

import uuid

from .mount_role import MountRole, parse_role


class Distribution:
    """РљР°Рє РёР· РѕРґРЅРѕР№ MountPoint РїРѕР»СѓС‡РёС‚СЊ РЅРµСЃРєРѕР»СЊРєРѕ MountLocation."""

    def __init__(
        self,
        count_x: int = 1,
        count_y: int = 1,
        spacing_x: float = 0.0,
        spacing_y: float = 0.0,
    ):
        self.count_x = max(1, int(count_x))
        self.count_y = max(1, int(count_y))
        self.spacing_x = max(0.0, float(spacing_x))
        self.spacing_y = max(0.0, float(spacing_y))

    def total_count(self) -> int:
        return self.count_x * self.count_y

    def is_simple(self) -> bool:
        """РџСЂРѕСЃС‚РѕРµ СЂР°СЃРїСЂРµРґРµР»РµРЅРёРµ - РѕРґРЅР° С‚РѕС‡РєР°, Р±РµР· СЂР°Р·РјРЅРѕР¶РµРЅРёСЏ."""
        return (
            self.count_x == 1
            and self.count_y == 1
            and self.spacing_x == 0.0
            and self.spacing_y == 0.0
        )

    def to_dict(self) -> dict:
        return {
            "count_x": self.count_x,
            "count_y": self.count_y,
            "spacing_x": self.spacing_x,
            "spacing_y": self.spacing_y,
        }

    @classmethod
    def from_dict(cls, d: dict) -> "Distribution":
        if not isinstance(d, dict):
            d = {}
        return cls(
            count_x=d.get("count_x", 1),
            count_y=d.get("count_y", 1),
            spacing_x=d.get("spacing_x", 0.0),
            spacing_y=d.get("spacing_y", 0.0),
        )


class MountPoint:
    """РћРїСЂРµРґРµР»РµРЅРёРµ С‚РѕС‡РєРё РєСЂРµРїР»РµРЅРёСЏ РІ Р»РѕРєР°Р»СЊРЅС‹С… РєРѕРѕСЂРґРёРЅР°С‚Р°С… Asset."""

    def __init__(
        self,
        id: str | None = None,
        role: MountRole | str | None = None,
        position: tuple[float, float] = (0.0, 0.0),
        distribution: Distribution | None = None,
        legacy_group_id: str | None = None,
        legacy_auto_rule: dict | None = None,
        anchor_mode: str = "absolute",
        anchor_x: float = 0.5,
        anchor_y: float = 0.5,
    ):
        self.id = id or self._generate_id()
        self.role: MountRole | None = parse_role(role)

        x, y = position
        self.position: tuple[float, float] = (float(x), float(y))

        self.distribution = (
            distribution if isinstance(distribution, Distribution)
            else Distribution()
        )

        # РџРѕР»СЏ РґР»СЏ РјРёРіСЂР°С†РёРё РёР· legacy (semantic_groups + auto_rule).
        # Р•СЃР»Рё MountPoint СЃРѕР·РґР°РЅ РІСЂСѓС‡РЅСѓСЋ вЂ” None.
        self.legacy_group_id: str | None = (
            str(legacy_group_id) if legacy_group_id else None
        )
        self.legacy_auto_rule: dict | None = (
            dict(legacy_auto_rule) if legacy_auto_rule else None
        )

        # V22: Anchor mode вЂ” РїСЂРёРІСЏР·РєР° Рє bbox РєРѕРЅС‚СѓСЂР°.
        # "absolute"     вЂ” position РІ Р»РѕРєР°Р»СЊРЅС‹С… РєРѕРѕСЂРґРёРЅР°С‚Р°С… Р°СЃСЃРµС‚Р° (РЅРµ РµРґРµС‚)
        # "relative_xy"  вЂ” position РІС‹С‡РёСЃР»СЏРµС‚СЃСЏ РєР°Рє xmin + ax*(xmax-xmin)
        self.anchor_mode: str = (
            str(anchor_mode) if anchor_mode else "absolute"
        )
        self.anchor_x: float = float(anchor_x)
        self.anchor_y: float = float(anchor_y)

    @staticmethod
    def _generate_id() -> str:
        return "mp_" + uuid.uuid4().hex[:8]

    # ------------------------------------------------------------

    def resolve(self, bbox=None) -> list["MountLocation"]:
        """Р Р°Р·РІРµСЂРЅСѓС‚СЊ MountPoint РІ СЃРїРёСЃРѕРє MountLocation.

        position - РїРµСЂРІР°СЏ location. count_x=3 -> СЂРѕРІРЅРѕ 3 РїРѕР·РёС†РёРё.

        bbox = (xmin, ymin, xmax, ymax) вЂ” РµСЃР»Рё Р·Р°РґР°РЅ Р anchor_mode="relative_xy",
        РІС‹С‡РёСЃР»СЏРµРј X/Y РєР°Рє РґРѕР»СЋ РѕС‚ bbox.
        """
        # V22: РѕРїСЂРµРґРµР»РёС‚СЊ Р±Р°Р·РѕРІСѓСЋ С‚РѕС‡РєСѓ
        if (
            self.anchor_mode == "relative_xy"
            and bbox is not None
            and len(bbox) == 4
        ):
            xmin, ymin, xmax, ymax = bbox
            # anchor_x: 0 = Р»РµРІС‹Р№ РєСЂР°Р№, 1 = РїСЂР°РІС‹Р№ (X РЅРµ РёРЅРІРµСЂС‚РёСЂРѕРІР°РЅ)
            x0 = xmin + self.anchor_x * (xmax - xmin)
            # anchor_y: 0 = Р’РР—РЈРђР›Р¬РќР«Р™ РЅРёР·, 1 = Р’РР—РЈРђР›Р¬РќР«Р™ РІРµСЂС….
            # VE РёСЃРїРѕР»СЊР·СѓРµС‚ Y-down (Р±РѕР»СЊС€Рµ Y вЂ” РЅРёР¶Рµ СЌРєСЂР°РЅ).
            # Р—РЅР°С‡РёС‚ ymin = РІРµСЂС…, ymax = РЅРёР·.
            # Р”Р»СЏ ay=1 (РІРµСЂС…) РЅСѓР¶РЅРѕ ymin в†’ y0 = ymax - ay*height.
            y0 = ymax - self.anchor_y * (ymax - ymin)
        else:
            x0, y0 = self.position

        result: list[MountLocation] = []
        sx = self.distribution.spacing_x
        sy = self.distribution.spacing_y

        for ix in range(self.distribution.count_x):
            for iy in range(self.distribution.count_y):
                px = x0 + ix * sx
                py = y0 + iy * sy
                result.append(
                    MountLocation(
                        mountpoint_id=self.id,
                        index_x=ix,
                        index_y=iy,
                        role=self.role,
                        position=(px, py),
                    )
                )
        return result

    def is_valid(self) -> bool:
        return bool(self.id)

    # ------------------------------------------------------------

    def to_dict(self) -> dict:
        from .mount_role import role_to_str
        d = {
            "id": self.id,
            "role": role_to_str(self.role),
            "position": [self.position[0], self.position[1]],
            "distribution": self.distribution.to_dict(),
        }
        if self.anchor_mode != "absolute":
            d["anchor_mode"] = self.anchor_mode
            d["anchor_x"] = self.anchor_x
            d["anchor_y"] = self.anchor_y
        if self.legacy_group_id is not None:
            d["legacy_group_id"] = self.legacy_group_id
        if self.legacy_auto_rule is not None:
            d["legacy_auto_rule"] = dict(self.legacy_auto_rule)
        return d

    @classmethod
    def from_dict(cls, d: dict) -> "MountPoint":
        if not isinstance(d, dict):
            raise ValueError("MountPoint.from_dict: dict expected")

        pos_raw = d.get("position") or [0.0, 0.0]
        if not isinstance(pos_raw, (list, tuple)) or len(pos_raw) < 2:
            pos_raw = [0.0, 0.0]

        legacy_ar = d.get("legacy_auto_rule")
        if legacy_ar is not None and not isinstance(legacy_ar, dict):
            legacy_ar = None

        return cls(
            id=d.get("id"),
            role=parse_role(d.get("role")),
            position=(pos_raw[0], pos_raw[1]),
            distribution=Distribution.from_dict(
                d.get("distribution") or {}
            ),
            legacy_group_id=d.get("legacy_group_id"),
            legacy_auto_rule=legacy_ar,
            anchor_mode=str(d.get("anchor_mode", "absolute")),
            anchor_x=float(d.get("anchor_x", 0.5)),
            anchor_y=float(d.get("anchor_y", 0.5)),
        )


class MountLocation:
    """Р’С‹С‡РёСЃР»РµРЅРЅР°СЏ РїРѕР·РёС†РёСЏ РґР»СЏ РєСЂРµРїР»РµРЅРёСЏ. РќР• СЃРѕС…СЂР°РЅСЏРµС‚СЃСЏ РІ Asset.

    РђРґСЂРµСЃСѓРµС‚СЃСЏ РїР°СЂРѕР№ (mountpoint_id, index_x, index_y).
    """

    __slots__ = (
        "mountpoint_id", "index_x", "index_y",
        "role", "position",
    )

    def __init__(
        self,
        mountpoint_id: str,
        index_x: int,
        index_y: int,
        role: MountRole | None,
        position: tuple[float, float],
    ):
        self.mountpoint_id = str(mountpoint_id)
        self.index_x = int(index_x)
        self.index_y = int(index_y)
        self.role: MountRole | None = role
        self.position: tuple[float, float] = (
            float(position[0]),
            float(position[1]),
        )

    def address(self) -> tuple[str, int, int]:
        """РЈРЅРёРєР°Р»СЊРЅС‹Р№ Р°РґСЂРµСЃ РІ СЃРёСЃС‚РµРјРµ."""
        return (self.mountpoint_id, self.index_x, self.index_y)

    def __repr__(self) -> str:
        return (
            f"MountLocation({self.mountpoint_id}"
            f"[{self.index_x},{self.index_y}] "
            f"role={self.role} pos={self.position})"
        )

    def __eq__(self, other) -> bool:
        if not isinstance(other, MountLocation):
            return NotImplemented
        return self.address() == other.address()


### attachment.py ###
"""
Attachment - СЃС‚СЂСѓРєС‚СѓСЂРёСЂРѕРІР°РЅРЅР°СЏ СЃРІСЏР·СЊ child-РєРѕРјРїРѕРЅРµРЅС‚Р° СЃ СЂРѕРґРёС‚РµР»РµРј.

target_type: "anchor" | "mountpoint"
target_id:   id anchor РёР»Рё id MountPoint
location:    {"x": int, "y": int} | None (С‚РѕР»СЊРєРѕ РґР»СЏ mountpoint)
child_anchor: РЅР°С€ anchor (bottom / top / ...)
"""

from __future__ import annotations


VALID_TARGET_TYPES = ("anchor", "mountpoint")


class Attachment:
    """РЎРІСЏР·СЊ РєРѕРјРїРѕРЅРµРЅС‚Р° СЃ СЂРѕРґРёС‚РµР»РµРј."""

    def __init__(
        self,
        target_type: str = "",
        target_id: str = "",
        location: dict | None = None,
        child_anchor: str = "",
    ):
        self.target_type: str = str(target_type or "")
        self.target_id: str = str(target_id or "")
        self.location: dict | None = (
            dict(location) if location else None
        )
        self.child_anchor: str = str(child_anchor or "")

    # ------------------------------------------------------------

    def is_valid(self) -> bool:
        if self.target_type not in VALID_TARGET_TYPES:
            return False
        if not self.target_id:
            return False
        if self.target_type == "mountpoint":
            if not isinstance(self.location, dict):
                return False
            if "x" not in self.location or "y" not in self.location:
                return False
        return True

    def is_anchor(self) -> bool:
        return self.target_type == "anchor"

    def is_mountpoint(self) -> bool:
        return self.target_type == "mountpoint"

    # ------------------------------------------------------------

    def to_dict(self) -> dict:
        return {
            "target_type": self.target_type,
            "target_id": self.target_id,
            "location": (
                dict(self.location) if self.location else None
            ),
            "child_anchor": self.child_anchor,
        }

    @classmethod
    def from_dict(cls, d: dict) -> "Attachment":
        if not isinstance(d, dict):
            raise ValueError("Attachment.from_dict: dict expected")

        loc = d.get("location")
        if loc is not None and not isinstance(loc, dict):
            loc = None

        return cls(
            target_type=d.get("target_type", ""),
            target_id=d.get("target_id", ""),
            location=loc,
            child_anchor=d.get("child_anchor", ""),
        )

    def __repr__(self) -> str:
        return (
            f"Attachment(type={self.target_type!r} "
            f"id={self.target_id!r} "
            f"loc={self.location} "
            f"child={self.child_anchor!r})"
        )


### migration.py ###
"""
РњРёРіСЂР°С†РёСЏ legacy semantic_groups + auto_rule -> MountPoint[].

РЎС‚Р°СЂС‹Рµ Р°СЃСЃРµС‚С‹ С…СЂР°РЅСЏС‚ С‚РѕС‡РєРё РєСЂРµРїР»РµРЅРёСЏ РІ semantic_groups СЃ auto_rule.
РќРѕРІР°СЏ РјРѕРґРµР»СЊ - MountPoint СЃ role, position, distribution.

РџСЂР°РІРёР»Р° РјРёРіСЂР°С†РёРё:
  - role = None (РЅРµ СѓРіР°РґС‹РІР°РµРј)
  - position = С†РµРЅС‚СЂРѕРёРґ СѓР·Р»РѕРІ-С€Р°Р±Р»РѕРЅРѕРІ РіСЂСѓРїРїС‹
  - distribution = default (count=1)
  - legacy_group_id / legacy_auto_rule СЃРѕС…СЂР°РЅСЏСЋС‚СЃСЏ РєР°Рє РµСЃС‚СЊ

Р¤СѓРЅРєС†РёСЏ РќР• РјСѓС‚РёСЂСѓРµС‚ Asset. Р’РѕР·РІСЂР°С‰Р°РµС‚ СЃРїРёСЃРѕРє MountPoint.
РРЅС‚РµРіСЂР°С†РёСЏ РІ Asset - РѕС‚РґРµР»СЊРЅС‹Р№ СЌС‚Р°Рї.
"""

from __future__ import annotations

from .mount_point import MountPoint, Distribution


def _collect_points_by_id(asset) -> dict[str, tuple[float, float]]:
    """РЎРѕР±СЂР°С‚СЊ РІСЃРµ node_id -> (x, y) РёР· РІРёРґРёРјС‹С… СЃР»РѕС‘РІ Asset.

    РўРѕС‚ Р¶Рµ РѕР±С…РѕРґ С‡С‚Рѕ РІ Asset.anchors() - С‚РѕР»СЊРєРѕ visible СЃР»РѕРё,
    fallback РЅР° asset.geometry РµСЃР»Рё СЃР»РѕС‘РІ РЅРµС‚.
    """
    pts: dict[str, tuple[float, float]] = {}

    layers = getattr(asset, "layers", None) or []
    geometries = []
    if layers:
        for layer in layers:
            if not getattr(layer, "visible", True):
                continue
            geometries.append(layer.geometry or {})
    if not geometries:
        geometries = [getattr(asset, "geometry", {}) or {}]

    for g in geometries:
        contour = g.get("contour", []) or []
        node_ids = g.get("node_ids", []) or []
        for i, nid in enumerate(node_ids):
            if i < len(contour):
                x, y = contour[i]
                pts[nid] = (float(x), float(y))

        extra_pts = g.get("extra_points", []) or []
        extra_ids = g.get("extra_node_ids", []) or []
        for i, nid in enumerate(extra_ids):
            if i < len(extra_pts):
                x, y = extra_pts[i]
                pts[nid] = (float(x), float(y))

    return pts


def _group_centroid(
    group, pts_by_id: dict[str, tuple[float, float]],
) -> tuple[float, float] | None:
    """Р¦РµРЅС‚СЂРѕРёРґ СѓР·Р»РѕРІ РіСЂСѓРїРїС‹. None РµСЃР»Рё РЅРё РѕРґРЅРѕРіРѕ СѓР·Р»Р° РЅРµС‚."""
    xs, ys = [], []
    for nid in group.node_ids:
        p = pts_by_id.get(nid)
        if p is not None:
            xs.append(p[0])
            ys.append(p[1])
    if not xs:
        return None
    return (sum(xs) / len(xs), sum(ys) / len(ys))


def migrate_asset(asset) -> list[MountPoint]:
    """РР·РІР»РµС‡СЊ MountPoint[] РёР· legacy semantic_groups СЃ auto_rule.

    Р“СЂСѓРїРїС‹ Р±РµР· auto_rule РёРіРЅРѕСЂРёСЂСѓСЋС‚СЃСЏ (СЌС‚Рѕ anchors / РїР°СЂР°РјРµС‚СЂРёР·Р°С†РёСЏ).
    Р’РѕР·РІСЂР°С‰Р°РµС‚ РЅРѕРІС‹Р№ СЃРїРёСЃРѕРє, Asset РЅРµ РјСѓС‚РёСЂСѓРµС‚СЃСЏ.
    """
    result: list[MountPoint] = []

    groups = getattr(asset, "semantic_groups", None) or {}
    if not groups:
        return result

    pts_by_id = _collect_points_by_id(asset)
    if not pts_by_id:
        return result

    for gid, group in groups.items():
        auto_rule = getattr(group, "auto_rule", None)
        if not auto_rule:
            continue

        centroid = _group_centroid(group, pts_by_id)
        if centroid is None:
            continue

        mp = MountPoint(
            role=None,
            position=centroid,
            distribution=Distribution(),
            legacy_group_id=gid,
            legacy_auto_rule=auto_rule,
        )
        result.append(mp)

    return result


def apply_migration(asset) -> int:
    """Р—Р°РїРѕР»РЅРёС‚СЊ asset.mountpoints СЂРµР·СѓР»СЊС‚Р°С‚РѕРј migrate_asset.

    Р’РѕР·РІСЂР°С‰Р°РµС‚ РєРѕР»РёС‡РµСЃС‚РІРѕ РґРѕР±Р°РІР»РµРЅРЅС‹С… MountPoint.
    Р•СЃР»Рё Сѓ asset СѓР¶Рµ РµСЃС‚СЊ РЅРµРїСѓСЃС‚РѕР№ mountpoints - РЅРµ С‚СЂРѕРіР°РµС‚.
    """
    existing = getattr(asset, "mountpoints", None)
    if existing:
        return 0

    migrated = migrate_asset(asset)
    if not migrated:
        return 0

    asset.mountpoints = migrated
    return len(migrated)


### asset.py (полностью) ###
"""
Asset вЂ” РїР°СЂР°РјРµС‚СЂРёР·РѕРІР°РЅРЅС‹Р№ Р°СЂС…РёС‚РµРєС‚СѓСЂРЅС‹Р№ РѕР±СЉРµРєС‚.

class Asset:

    def __init__(
        self,
        asset_id: str | None = None,
        name: str = "РќРѕРІС‹Р№ Р°СЃСЃРµС‚",
        type_: str = "other",
        geometry: dict | None = None,
        components: dict | None = None,
        semantic_groups: dict | None = None,
        parameters: dict | None = None,
        generation_rules: dict | None = None,
    ):
        self.id = asset_id or self._generate_id()
        self.name = name
        self.type = type_ if type_ in ASSET_TYPE_IDS else "other"
        self.geometry = geometry or self._empty_geometry()
        # РЎСЃС‹Р»РєРё РЅР° РґСЂСѓРіРёРµ Asset'С‹ (СЃРѕСЃС‚Р°РІРЅРѕР№ Asset).
        # Р›РёР±Рѕ geometry СЃ РєРѕРЅС‚СѓСЂРѕРј, Р»РёР±Рѕ components вЂ” РЅРѕ РЅРµ РѕР±Р°.
        self.components: dict[str, Component] = components or {}
        self.visibility_rules: dict[str, VisibilityRule] = {}
        self.semantic_groups: dict[str, SemanticGroup] = semantic_groups or {}
        self.parameters = parameters or {}
        self.reference_image: ReferenceImage | None = None
        self.generation_rules = generation_rules or self._default_rules()
        # V13: РѕРїРѕСЂРЅР°СЏ Р»РёРЅРёСЏ (ground line) РґР»СЏ РєРѕРјРїРѕР·РёС‚Р°.
        # ground_line_y = None вЂ” Р»РёРЅРёРё РЅРµС‚.

        self.ground_line_y: float | None = None

        # V16: СЃР»РѕРё. РџРѕРєР° С‚РѕР»СЊРєРѕ РјРѕРґРµР»СЊ вЂ” UI РїРѕР·Р¶Рµ.
        # РџСЂРё Р·Р°РіСЂСѓР·РєРµ СЃС‚Р°СЂРѕРіРѕ JSON СЃРѕР·РґР°С‘С‚СЃСЏ РѕРґРёРЅ СЃР»РѕР№ РёР· geometry.

        # РїСЂРё Р·Р°РіСЂСѓР·РєРµ СЃС‚Р°СЂРѕРіРѕ JSON Р±РµР· РїРѕР»СЏ "mountpoints".
        self.mountpoints: list[MountPoint] = []

    @staticmethod
    def _generate_id() -> str:
        return uuid.uuid4().hex[:12]

    @staticmethod
    def _empty_geometry() -> dict:
        return {
            "contour": [],
            "closed": True,
            "units": "m",
            "node_ids": [],
            "groups": {},
            "extra_edges": [],
            "arcs": [],
            "extra_points": [],
            "extra_node_ids": [],
        }

    @staticmethod
    def _default_rules() -> dict:
        return {
            "allow_width_variation": False,
            "allow_height_variation": False,
            "irregularity": 0.0,
            "repeatable": False,
        }

    @classmethod
    def from_contour(
        cls,
        contour: VectorContour,
        name: str = "РќРѕРІС‹Р№ Р°СЃСЃРµС‚",
        type_: str = "other",
        asset_id: str | None = None,
    ) -> "Asset":
        """РЎРѕР·РґР°С‚СЊ Asset РёР· СЃСѓС‰РµСЃС‚РІСѓСЋС‰РµРіРѕ VectorContour.

        РђРІС‚РѕРјР°С‚РёС‡РµСЃРєРё СЃРѕР·РґР°С‘С‚ node_ids Рё С‚РµС…РЅРёС‡РµСЃРєРёРµ РіСЂСѓРїРїС‹ СЂС‘Р±РµСЂ.
        asset_id вЂ” РµСЃР»Рё РїРµСЂРµРґР°РЅ, СЃРѕС…СЂР°РЅСЏРµС‚СЃСЏ (РґР»СЏ Save РїРѕРІРµСЂС…).
        """
        points = list(contour.points)
        n = len(points)

        # РСЃРїРѕР»СЊР·СѓРµРј node_ids РёР· РєРѕРЅС‚СѓСЂР°, РµСЃР»Рё РѕРЅРё РµСЃС‚СЊ Рё СЃРѕРІРїР°РґР°СЋС‚
        # РїРѕ РґР»РёРЅРµ. РРЅР°С‡Рµ вЂ” РіРµРЅРµСЂРёСЂСѓРµРј Р·Р°РЅРѕРІРѕ.
        node_ids = list(contour.node_ids)
        if len(node_ids) != n:
            node_ids = [f"n_{i:03d}" for i in range(n)]

        groups: dict[str, list[str]] = {}
        if n >= 2:
            segments = range(n) if contour.closed else range(n - 1)
            for i in segments:
                a = node_ids[i]
                b = node_ids[(i + 1) % n]
                groups[f"edge_{i:03d}"] = [a, b]

        # Extra-СѓР·Р»С‹ Рё extra-edges СЃРѕС…СЂР°РЅСЏСЋС‚СЃСЏ РєР°Рє РµСЃС‚СЊ (РѕРЅРё
        # СЃСЃС‹Р»Р°СЋС‚СЃСЏ РЅР° node_ids РёР· main РёР»Рё extra).
        extra_points = list(getattr(contour, "extra_points", []))
        extra_node_ids = list(
            getattr(contour, "extra_node_ids", [])
        )

        # РЎРёРЅС…СЂРѕРЅРёР·Р°С†РёСЏ (РЅР° РІСЃСЏРєРёР№ СЃР»СѓС‡Р°Р№)
        while len(extra_node_ids) < len(extra_points):
            extra_node_ids.append(
                f"e_{len(extra_node_ids):03d}"
            )

        valid_all_ids = set(node_ids) | set(extra_node_ids)

        geometry = {
            "contour": [[float(x), float(y)] for (x, y) in points],
            "closed": bool(contour.closed),
            "units": "m",
            "node_ids": node_ids,
            "groups": groups,
            "extra_edges": [
                [a, b]
                for (a, b) in getattr(contour, "extra_edges", [])
                if a in valid_all_ids and b in valid_all_ids
            ],
            "extra_points": [
                [float(x), float(y)] for (x, y) in extra_points
            ],
            "extra_node_ids": extra_node_ids,
            "arcs": [
                [a, b, float(v)]
                for (a, b), v in getattr(contour, "arcs", {}).items()
                if a in valid_all_ids and b in valid_all_ids
                and abs(v) > 1e-9
            ],
        }

        return cls(
            asset_id=asset_id,
            name=name,
            type_=type_,
            geometry=geometry,
        )

            "ground_line_visible": self.ground_line_visible,
            "layers": [l.to_dict() for l in self.layers],
            "mountpoints": [

    @classmethod
    def from_dict(cls, d: dict) -> "Asset":
        # V22: Р±С‹Р»Рѕ Р»Рё РїРѕР»Рµ mountpoints РІ РёСЃС…РѕРґРЅРѕРј JSON.
        # Р•СЃР»Рё РЅРµС‚ вЂ” СЌС‚Рѕ СЃС‚Р°СЂС‹Р№ Р°СЃСЃРµС‚, РјРёРіСЂРёСЂСѓРµРј legacy РІ РєРѕРЅС†Рµ.
        _has_mountpoints_field = "mountpoints" in d

        groups_raw = d.get("semantic_groups") or {}
        groups: dict[str, SemanticGroup] = {}
        for gid, gdict in groups_raw.items():
            try:
                groups[gid] = SemanticGroup.from_dict(gdict)
            except Exception:
                continue

        params_raw = d.get("parameters") or {}
        params: dict[str, Parameter] = {}
        if isinstance(params_raw, dict):
            for pid, pdict in params_raw.items():
                if not isinstance(pdict, dict):
                    continue
                try:
                    params[pid] = Parameter.from_dict(pdict)
                except Exception:
                    continue

        rules_raw = d.get("visibility_rules") or {}
        rules: dict[str, VisibilityRule] = {}
        if isinstance(rules_raw, dict):
            for rid, rdict in rules_raw.items():
                if not isinstance(rdict, dict):
                    continue
                try:
                    rules[rid] = VisibilityRule.from_dict(rdict)
                except Exception:
                    continue

        comp_raw = d.get("components") or {}
        comps: dict[str, Component] = {}
        if isinstance(comp_raw, dict):
            for cid, cdict in comp_raw.items():
                if not isinstance(cdict, dict):
                    continue
                try:
                    comps[cid] = Component.from_dict(cdict)
                except Exception:
                    continue

        ref_raw = d.get("reference_image")
        ref: ReferenceImage | None = None
        if isinstance(ref_raw, dict):
            try:
                ref = ReferenceImage.from_dict(ref_raw)
            except Exception:
                ref = None

        asset = cls(
            asset_id=d.get("id"),
            name=d.get("name", "РќРѕРІС‹Р№ Р°СЃСЃРµС‚"),
            type_=d.get("type", "other"),
            geometry=d.get("geometry") or cls._empty_geometry(),
            components=comps,
            semantic_groups=groups,
            parameters=params,
            generation_rules=d.get("generation_rules") or cls._default_rules(),
        )
        asset.reference_image = ref
        asset.visibility_rules = rules

        # V13: ground line
        g_y = d.get("ground_line_y")
        if isinstance(g_y, (int, float)):
            asset.ground_line_y = float(g_y)
        else:
            asset.ground_line_y = None
        asset.ground_line_visible = bool(
            d.get("ground_line_visible", True)
        )

        # V16: СЃР»РѕРё
        layers_raw = d.get("layers") or []
        if layers_raw:
            for ldict in layers_raw:
                if not isinstance(ldict, dict):
                    continue
                try:
                    asset.layers.append(VectorLayer.from_dict(ldict))
                except Exception:
                    continue
        else:
            # РђРІС‚РѕРјРёРіСЂР°С†РёСЏ: СЃС‚Р°СЂС‹Р№ JSON Р±РµР· layers вЂ”
            # СЃРѕР·РґР°С‚СЊ РѕРґРёРЅ СЃР»РѕР№ РёР· geometry.
            asset.layers.append(VectorLayer(
                name="РћСЃРЅРѕРІРЅРѕР№",
                geometry=dict(asset.geometry),
            ))

        # V22 (Mounting): С‡РёС‚Р°РµРј mountpoints РёР· JSON
        mps_raw = d.get("mountpoints")
        if isinstance(mps_raw, list):
            for mdict in mps_raw:
                if not isinstance(mdict, dict):
                    continue
                try:
                    asset.mountpoints.append(
                        MountPoint.from_dict(mdict)
                    )
                except Exception:
                    continue

        # V22: РµСЃР»Рё РїРѕР»СЏ mountpoints РЅРµ Р±С‹Р»Рѕ вЂ” РјРёРіСЂРёСЂСѓРµРј legacy
        # (semantic_groups + auto_rule -> MountPoint).
        # РњРёРіСЂР°С†РёСЏ РёРґРµРјРїРѕС‚РµРЅС‚РЅР° Рё РЅРµ РјСѓС‚РёСЂСѓРµС‚ semantic_groups.
        if not _has_mountpoints_field:
            try:
                apply_migration(asset)
            except Exception:
                pass

        return asset

    def points(self) -> list[tuple[float, float]]:
        return [(float(p[0]), float(p[1])) for p in self.geometry.get("contour", [])]

    def is_closed(self) -> bool:
        return bool(self.geometry.get("closed", True))

    def node_count(self) -> int:

    def group_count(self) -> int:
        return len(self.geometry.get("groups", {}))

    def extra_edges(self) -> list[tuple[str, str]]:
        """Р’РµСЂРЅСѓС‚СЊ extra_edges, РЅРѕСЂРјР°Р»РёР·РѕРІР°РІ СЃС‚Р°СЂС‹Р№ index-С„РѕСЂРјР°С‚.

        - int-РёРЅРґРµРєСЃС‹ РїСЂРµРІСЂР°С‰Р°СЋС‚СЃСЏ РІ node_ids РїРѕ node_ids-СЃРїРёСЃРєСѓ
        - РѕС‚Р±СЂР°СЃС‹РІР°СЋС‚СЃСЏ РїР°СЂС‹ СЃ РЅРµСЃСѓС‰РµСЃС‚РІСѓСЋС‰РёРјРё id
        - РѕС‚Р±СЂР°СЃС‹РІР°СЋС‚СЃСЏ self-РїР°СЂС‹
        - РѕС‚Р±СЂР°СЃС‹РІР°СЋС‚СЃСЏ РґСѓР±Р»РёРєР°С‚С‹

        РџРѕСЃР»Рµ РІС‹Р·РѕРІР° `geometry["extra_edges"]` РїРµСЂРµР·Р°РїРёСЃС‹РІР°РµС‚СЃСЏ
        РЅРѕСЂРјР°Р»РёР·РѕРІР°РЅРЅС‹Рј СЃРїРёСЃРєРѕРј.
        """
        node_ids = list(self.geometry.get("node_ids", []))
        extra_ids = list(self.geometry.get("extra_node_ids", []))
        valid = set(node_ids) | set(extra_ids)

        # РЎРєР»РµРµРЅРЅС‹Р№ СЃРїРёСЃРѕРє РґР»СЏ backward compat int в†’ node_id.
        # РџРѕСЂСЏРґРѕРє: СЃРЅР°С‡Р°Р»Р° main, РїРѕС‚РѕРј extra вЂ” СЃРѕРѕС‚РІРµС‚СЃС‚РІСѓРµС‚
        # РїРѕСЂСЏРґРєСѓ, РІ РєРѕС‚РѕСЂРѕРј РєРѕРіРґР°-С‚Рѕ РјРѕРіР»Рё РёРґС‚Рё РёРЅРґРµРєСЃС‹.
        all_ids = node_ids + extra_ids

        result: list[tuple[str, str]] = []

        for raw in self.geometry.get("extra_edges", []) or []:
            if not isinstance(raw, (list, tuple)):
                continue
            if len(raw) != 2:
                continue

            a, b = raw

            if isinstance(a, int) and 0 <= a < len(all_ids):
                a = all_ids[a]
            if isinstance(b, int) and 0 <= b < len(all_ids):
                b = all_ids[b]

            if not isinstance(a, str) or not isinstance(b, str):
                continue
            if a == b:
                continue
            if a not in valid or b not in valid:
                continue

            if any(
                frozenset(edge) == frozenset((a, b))
                for edge in result
            ):
                continue

            result.append((a, b))

        # РќРѕСЂРјР°Р»РёР·СѓРµРј geometry СЃСЂР°Р·Сѓ
        self.geometry["extra_edges"] = [
            [a, b] for a, b in result
        ]

        return result

    def anchors(self) -> dict[str, tuple[float, float]]:
        """РўРѕС‡РєРё РєСЂРµРїР»РµРЅРёСЏ РёР· РіСЂСѓРїРї СЃ РїСЂРµС„РёРєСЃРѕРј `anchor_`.

        Position = С†РµРЅС‚СЂРѕРёРґ СѓР·Р»РѕРІ РіСЂСѓРїРїС‹. Р•СЃР»Рё СѓР·Р»С‹ РЅРµ РЅР°Р№РґРµРЅС‹
        (РёР»Рё РіСЂСѓРїРїР° РїСѓСЃС‚Р°) вЂ” anchor РїСЂРѕРїСѓСЃРєР°РµС‚СЃСЏ.

        Р’РѕР·РІСЂР°С‰Р°РµС‚ {tag: (x, y)}. tag вЂ” РёРјСЏ Р±РµР· РїСЂРµС„РёРєСЃР°.
        """
        result: dict[str, tuple[float, float]] = {}

        # V16: С‚РѕС‡РєРё РёР· Р’РЎР•РҐ РІРёРґРёРјС‹С… СЃР»РѕС‘РІ (РЅРµ С‚РѕР»СЊРєРѕ Р°РєС‚РёРІРЅРѕРіРѕ).
        # Р Р°РЅСЊС€Рµ anchors С‚РµСЂСЏР»РёСЃСЊ, РµСЃР»Рё anchor_bottom Р±С‹Р»
        # РІ РЅРµР°РєС‚РёРІРЅРѕРј СЃР»РѕРµ РѕРєРЅР°.
        pts_by_id: dict[str, tuple[float, float]] = {}

        layers = getattr(self, "layers", None) or []
        layer_geometries = []
        if layers:
            for layer in layers:
                if not getattr(layer, "visible", True):
                    continue
                layer_geometries.append(layer.geometry or {})
        if not layer_geometries:
            # РЎС‚Р°СЂС‹Р№ С„РѕСЂРјР°С‚ вЂ” РѕРґРёРЅ СЃР»РѕР№ РІ geometry
            layer_geometries = [self.geometry]

        for g in layer_geometries:
            contour = g.get("contour", [])
            node_ids = g.get("node_ids", [])
            for i, nid in enumerate(node_ids):
                if i < len(contour):
                    x, y = contour[i]
                    pts_by_id[nid] = (float(x), float(y))
            extra_pts = g.get("extra_points", [])
            extra_ids = g.get("extra_node_ids", [])
            for i, nid in enumerate(extra_ids):
                if i < len(extra_pts):
