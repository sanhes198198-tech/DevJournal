class ComponentItem(QGraphicsObject):
    """QGraphicsObject-РѕР±С‘СЂС‚РєР° РґР»СЏ Component."""

    # Р­РјРёС‚РёС‚СЃСЏ РїРѕСЃР»Рµ Р·Р°РІРµСЂС€РµРЅРёСЏ drag РёР»Рё РїРѕСЃР»Рµ apply_from_component
    moved = Signal()
    # Р”РІРѕР№РЅРѕР№ РєР»РёРє РїРѕ composite вЂ” СЃРёРіРЅР°Р» В«РІРѕР№С‚РёВ» СЃ asset_id
    enter_requested = Signal(str)
    # (comp_id, old_x, old_y, new_x, new_y) вЂ” РїРѕСЃР»Рµ drag
    drag_finished = Signal(str, float, float, float, float)
    param_changed = Signal(str, str, float)

    MAX_DEPTH = 8

    def __init__(
        self, component, asset=None, registry=None, parent=None,
    ):
        super().__init__(parent)
        self._component = component
        self._asset = asset
        self._registry = registry
        self._path = QPainterPath()
        self._stroke_path = QPainterPath()  # V16: СЃР»РѕРё СЃ fillable=False
        self._extra_path = QPainterPath()

        # РЎРјРµС‰РµРЅРёРµ РєРѕРЅС‚СѓСЂР°, С‡С‚РѕР±С‹ РµРіРѕ Р¦Р•РќРўР  Р±С‹Р» РІ (0,0).
        # РРЅР°С‡Рµ Р±Р°С€РЅСЏ СЂРёСЃСѓРµС‚СЃСЏ СЃР±РѕРєСѓ РѕС‚ origin.
        self._offset_x = 0.0
        self._offset_y = 0.0
        self._center_x = 0.0
        self._center_y = 0.0
        # Snap
        self._highlighted_anchor: str | None = None
        self._snap_partner = None
        # V13: С„Р»Р°Рі С‡С‚Рѕ snap Р±С‹Р» Рє ground line (РЅРµ РєРѕРјРїРѕРЅРµРЅС‚Сѓ)
        self._snap_to_ground = False
        # Undo: РїРѕР·РёС†РёСЏ РґРѕ drag
        self._drag_old_pos = None
        # On-canvas handles
        self._handles: dict = {}

        self.setFlag(QGraphicsItem.GraphicsItemFlag.ItemIsSelectable, True)
        self.setFlag(QGraphicsItem.GraphicsItemFlag.ItemIsMovable, True)
        self.setFlag(
            QGraphicsItem.GraphicsItemFlag.ItemSendsGeometryChanges,
            True,
        )
        self.setZValue(3.0)
        self.setCacheMode(QGraphicsItem.CacheMode.NoCache)

        # РЎРЅР°Рї Рє 0.1 Рј РїСЂРё РїРµСЂРµРјРµС‰РµРЅРёРё
        self._snap_step = 0.1

        self._rebuild_paths()
        self._apply_transform()

    # ------------------------------------------------------------

    @property
    def component(self):
        return self._component

    def set_asset(self, asset) -> None:
        self._asset = asset
        self.prepareGeometryChange()
        self._rebuild_paths()
        self.update()

    # ------------------------------------------------------------

    def _apply_transform(self) -> None:
        """РџРѕР·РёС†РёСЏ/РїРѕРІРѕСЂРѕС‚/РјР°СЃС€С‚Р°Р±/layer РёР· Component."""
        c = self._component
        self.setPos(c.x, c.y)
        self.setRotation(-c.rotation)
        self.setScale(c.scale)
        # Z РѕС‚ СЃР»РѕСЏ: 100 + layer, С‡С‚РѕР±С‹ РЅРµ РїРµСЂРµСЃРµРєР°С‚СЊСЃСЏ СЃ UI
        # V9-РґРѕРї: РѕРєРЅР° +10 С‡С‚РѕР±С‹ РєР»РёРє РїРѕ РЅРёРј Р»РѕРІРёР»СЃСЏ РїРѕРІРµСЂС… СЃС‚РµРЅ
        self.setZValue(self._compute_z())

        # V12: РµСЃР»Рё РєРѕРјРїРѕРЅРµРЅС‚ Р·Р°Р±Р»РѕРєРёСЂРѕРІР°РЅ вЂ” РЅРµР»СЊР·СЏ С‚Р°С‰РёС‚СЊ
        locked = bool(getattr(c, "locked", False))
        self.setFlag(
            QGraphicsItem.GraphicsItemFlag.ItemIsMovable,
            not locked,
        )

    def _compute_z(self) -> float:
        """Z РёР· layer + Р±РѕРЅСѓСЃ РѕРєРЅР°Рј (С‡С‚РѕР±С‹ РѕРєРЅР° Р±С‹Р»Рё РїРѕРІРµСЂС… СЃС‚РµРЅ)."""
        c = self._component
        base = 100.0 + float(getattr(c, "layer", 0))
        asset_type = ""
        if self._asset is not None:
            asset_type = getattr(self._asset, "type", "")
        if asset_type == "window":
            base += 10.0
        return base

    def _clear_handles(self) -> None:
        for h in self._handles.values():
            h.setParentItem(None)
            if h.scene() is not None:
                h.scene().removeItem(h)
        self._handles.clear()

    def _rebuild_handles(self) -> None:
        self._clear_handles()
        if not self.isSelected():
            return
        if self._asset is None:
            return

        params = getattr(self._asset, "parameters", None) or {}
        if not params:
            return

        r = self._path.boundingRect()
        if not self._extra_path.isEmpty():
            er = self._extra_path.boundingRect()
            r = r.united(er)
        if r.isEmpty():
            return

        overrides = getattr(self._component, "param_overrides", None) or {}

        for p in params.values():
            if p.name == "height":
                base_v = float(p.value)
                cur_v = float(overrides.get("height", base_v))
                h = ParamHandleItem(
                    "height", "v", base_v, cur_v, parent=self,
                )
                h.setPos(r.right() + 0.8, r.center().y())
                h.value_changed.connect(self._on_handle_changed)
                self._handles["height"] = h
            elif p.name == "width":
                base_v = float(p.value)
                cur_v = float(overrides.get("width", base_v))
                h = ParamHandleItem(
                    "width", "h", base_v, cur_v, parent=self,
                )
                h.setPos(r.center().x(), r.bottom() + 0.8)
                h.value_changed.connect(self._on_handle_changed)
                self._handles["width"] = h

            # V15: СЂСѓС‡РєРё СЂР°СЃС‚СЏР¶РµРЅРёСЏ СЃС‚РѕСЂРѕРЅ
            elif p.name == "extend_right":
                base_v = float(p.value)
                cur_v = float(overrides.get("extend_right", base_v))
                h = ParamHandleItem(
                    "extend_right", "v", base_v, cur_v, parent=self,
                )
                # РЎРїСЂР°РІР° РѕС‚ bbox, РґР°Р»СЊС€Рµ С‡РµРј height
                h.setPos(r.right() + 2.5, r.center().y())
                h.value_changed.connect(self._on_handle_changed)
                self._handles["extend_right"] = h
            elif p.name == "extend_left":
                base_v = float(p.value)
                cur_v = float(overrides.get("extend_left", base_v))
                h = ParamHandleItem(
                    "extend_left", "v", base_v, cur_v, parent=self,
                )
                h.setPos(r.left() - 0.8, r.center().y())
                h.value_changed.connect(self._on_handle_changed)
                self._handles["extend_left"] = h
            elif p.name == "extend_top":
                base_v = float(p.value)
                cur_v = float(overrides.get("extend_top", base_v))
                h = ParamHandleItem(
                    "extend_top", "h", base_v, cur_v, parent=self,
                )
                h.setPos(r.center().x(), r.top() - 0.8)
                h.value_changed.connect(self._on_handle_changed)
                self._handles["extend_top"] = h
            elif p.name == "extend_bottom":
                base_v = float(p.value)
                cur_v = float(overrides.get("extend_bottom", base_v))
                h = ParamHandleItem(
                    "extend_bottom", "h", base_v, cur_v, parent=self,
                )
                # РЎРЅРёР·Сѓ РѕС‚ bbox, РґР°Р»СЊС€Рµ С‡РµРј width
                h.setPos(r.center().x(), r.bottom() + 2.5)
                h.value_changed.connect(self._on_handle_changed)
                self._handles["extend_bottom"] = h

    def _on_handle_changed(self, name: str, value: float) -> None:
        self.param_changed.emit(self._component.id, name, value)

    def _rebuild_paths(self) -> None:
        self._path = QPainterPath()
        self._stroke_path = QPainterPath()
        self._extra_path = QPainterPath()

        if self._asset is None:
            return

        # V16: СЃР±РѕСЂРєР° РІРѕР·РІСЂР°С‰Р°РµС‚ 4 РїСѓС‚Рё вЂ”
        # fill (Р·Р°Р»РёРІР°РµС‚СЃСЏ), stroke (С‚РѕР»СЊРєРѕ РєРѕРЅС‚СѓСЂ),
        # extra (СЂС‘Р±СЂР°), center.
        overrides = getattr(self._component, "param_overrides", None) or {}
        fill, stroke, extra, center = self._collect_asset_paths(
            self._asset, depth=0, param_overrides=overrides,
        )
        self._path = fill
        self._stroke_path = stroke
        self._extra_path = extra
        self._center_x, self._center_y = center

    def _collect_asset_paths(
        self, asset, depth: int, param_overrides: dict | None = None,
    ) -> tuple[
        QPainterPath, QPainterPath, QPainterPath, tuple[float, float]
    ]:
        """V16: СЃРѕР±СЂР°С‚СЊ РїСѓС‚Рё РёР· asset'Р° СЂРµРєСѓСЂСЃРёРІРЅРѕ.

        Р’РѕР·РІСЂР°С‰Р°РµС‚ (fill_path, stroke_path, extra_path, (cx, cy)):
          - fill_path   вЂ” СЃР»РѕРё СЃ fillable=True (Р·Р°Р»РёРІР°СЋС‚СЃСЏ С‚РµРєСЃС‚СѓСЂРѕР№)
          - stroke_path вЂ” СЃР»РѕРё СЃ fillable=False (С‚РѕР»СЊРєРѕ РєРѕРЅС‚СѓСЂ)
          - extra_path  вЂ” extra-СЂС‘Р±СЂР° РІСЃРµС… СЃР»РѕС‘РІ
          - center      вЂ” С†РµРЅС‚СЂ bbox РїРµСЂРІРѕРіРѕ СЃР»РѕСЏ (РґР»СЏ anchor'РѕРІ)
        """
        fill = QPainterPath()
        stroke = QPainterPath()
        extra = QPainterPath()

        if asset is None:
            return fill, stroke, extra, (0.0, 0.0)

        if depth > self.MAX_DEPTH:
            return fill, stroke, extra, (0.0, 0.0)

        # V16: СЃРїРёСЃРѕРє СЃР»РѕС‘РІ. РЎС‚Р°СЂС‹Рµ Р°СЃСЃРµС‚С‹ Р±РµР· layers в†’
        # РµРґРёРЅСЃС‚РІРµРЅРЅС‹Р№ fillable-СЃР»РѕР№ РёР· asset.geometry.
        layer_items: list[tuple[dict, bool]] = []
        asset_layers = getattr(asset, "layers", None) or []
        if asset_layers:
            for layer in asset_layers:
                if not getattr(layer, "visible", True):
                    continue
                layer_items.append((
                    layer.geometry or {},
                    bool(getattr(layer, "fillable", True)),
                ))
        else:
            layer_items.append((asset.geometry, True))

        if not layer_items:
            # Р’СЃРµ СЃР»РѕРё СЃРєСЂС‹С‚С‹
            return fill, stroke, extra, (0.0, 0.0)

        # Р”Р»СЏ С†РµРЅС‚СЂР° Р±РµСЂС‘Рј РџР•Р Р’Р«Р™ СЃР»РѕР№
        first_geo, _ = layer_items[0]
        orig_contour = list(first_geo.get("contour", []))

        for layer_geo, fillable in layer_items:
            contour = list(layer_geo.get("contour", []))
            node_ids = list(layer_geo.get("node_ids", []))
            extra_pts = list(layer_geo.get("extra_points", []))
            extra_ids = list(layer_geo.get("extra_node_ids", []))
            extra_edges = layer_geo.get("extra_edges", [])

            # --- Override РїР°СЂР°РјРµС‚СЂРѕРІ ---
            if param_overrides:
                params = getattr(asset, "parameters", None) or {}
                sem_groups = getattr(asset, "semantic_groups", None) or {}
                for p in params.values():
                    ov = param_overrides.get(p.name)
                    if ov is None:
                        continue
                    delta_shift = float(ov) - float(p.value)
                    if abs(delta_shift) < 1e-12:
                        continue
                    node_delta = compute_delta_for_parameter(
                        p, delta_shift, sem_groups,
                    )
                    contour = apply_delta_to_points(
                        contour, node_ids, node_delta,
                    )
                    extra_pts = apply_delta_to_points(
                        extra_pts, extra_ids, node_delta,
                    )

            # --- Arcs ---
            arcs_map: dict[tuple, float] = {}
            for arc in layer_geo.get("arcs", []) or []:
                if not isinstance(arc, (list, tuple)) or len(arc) < 3:
                    continue
                try:
                    bv = float(arc[2])
                except (TypeError, ValueError):
                    continue
                if abs(bv) < 1e-9:
                    continue
                arcs_map[(str(arc[0]), str(arc[1]))] = bv

            def _get_bulge(a_id, b_id, _m=arcs_map):
                if not a_id or not b_id:
                    return 0.0
                v = _m.get((a_id, b_id))
                if v is not None:
                    return v
                v = _m.get((b_id, a_id))
                if v is not None:
                    return v
                return 0.0

            def _append_segment(path, p1, p2, a_id, b_id):
                bulge = _get_bulge(a_id, b_id)
                if abs(bulge) < 1e-9:
                    path.lineTo(float(p2[0]), float(p2[1]))
                    return
                x1, y1 = float(p1[0]), float(p1[1])
                x2, y2 = float(p2[0]), float(p2[1])
                dx = x2 - x1
                dy = y2 - y1
                L = (dx * dx + dy * dy) ** 0.5
                if L < 1e-9:
                    path.lineTo(x2, y2)
                    return
                mx = (x1 + x2) * 0.5
                my = (y1 + y2) * 0.5
                nx = -dy / L
                ny = dx / L
                k = 2.0 * bulge * L
                cx = mx + nx * k
                cy = my + ny * k
                path.quadTo(cx, cy, x2, y2)

            # --- Р¦РµР»РµРІРѕР№ path РїРѕ fillable ---
            target = fill if fillable else stroke

            # --- Main РєРѕРЅС‚СѓСЂ СЌС‚РѕРіРѕ СЃР»РѕСЏ ---
            n_pts = len(contour)
            n_ids = len(node_ids)
            if n_pts >= 2:
                x0, y0 = contour[0]
                target.moveTo(float(x0), float(y0))
                for i in range(1, n_pts):
                    p1 = contour[i - 1]
                    p2 = contour[i]
                    a_id = node_ids[i - 1] if (i - 1) < n_ids else None
                    b_id = node_ids[i] if i < n_ids else None
                    _append_segment(target, p1, p2, a_id, b_id)
                if layer_geo.get("closed", True):
                    p1 = contour[n_pts - 1]
                    p2 = contour[0]
                    a_id = (
                        node_ids[n_pts - 1]
                        if (n_pts - 1) < n_ids else None
                    )
                    b_id = node_ids[0] if n_ids > 0 else None
                    _append_segment(target, p1, p2, a_id, b_id)
                    target.closeSubpath()

            # --- Extra edges СЌС‚РѕРіРѕ СЃР»РѕСЏ в†’ РІ РѕР±С‰РёР№ extra ---
            pos_map: dict[str, tuple[float, float]] = {}
            for i, nid in enumerate(node_ids):
                if i < len(contour):
                    x, y = contour[i]
                    pos_map[nid] = (float(x), float(y))
            for i, nid in enumerate(extra_ids):
                if i < len(extra_pts):
                    x, y = extra_pts[i]
                    pos_map[nid] = (float(x), float(y))

            for edge in extra_edges:
                if not isinstance(edge, (list, tuple)) or len(edge) != 2:
                    continue
                a_id, b_id = edge
                pa = pos_map.get(a_id)
                pb = pos_map.get(b_id)
                if pa is None or pb is None:
                    continue
                extra.moveTo(pa[0], pa[1])
                extra.lineTo(pb[0], pb[1])

        # --- Р’Р»РѕР¶РµРЅРЅС‹Рµ РєРѕРјРїРѕРЅРµРЅС‚С‹ (СЂРµРєСѓСЂСЃРёСЏ) ---
        comps = getattr(asset, "components", {})
        if comps and self._registry is not None:
            from PySide6.QtGui import QTransform
            for comp in comps.values():
                sub = self._registry.get(comp.asset_id)
                if sub is None:
                    continue
                sub_fill, sub_stroke, sub_extra, _ = (
                    self._collect_asset_paths(
                        sub, depth + 1,
                        param_overrides=comp.param_overrides,
                    )
                )
                t = QTransform()
                t.translate(comp.x, comp.y)
                t.rotate(-comp.rotation)
                t.scale(comp.scale, comp.scale)
                if not sub_fill.isEmpty():
                    fill.addPath(t.map(sub_fill))
                if not sub_stroke.isEmpty():
                    stroke.addPath(t.map(sub_stroke))
                if not sub_extra.isEmpty():
                    extra.addPath(t.map(sub_extra))

        # --- Р¦РµРЅС‚СЂРёСЂРѕРІР°РЅРёРµ РїРѕ bbox РїРµСЂРІРѕРіРѕ СЃР»РѕСЏ ---
        if orig_contour:
            xs = [float(p[0]) for p in orig_contour]
            ys = [float(p[1]) for p in orig_contour]
            cx = (min(xs) + max(xs)) / 2
            cy = (min(ys) + max(ys)) / 2
            from PySide6.QtGui import QTransform
            shift = QTransform()
            shift.translate(-cx, -cy)
            fill = shift.map(fill)
            stroke = shift.map(stroke)
            extra = shift.map(extra)
            return fill, stroke, extra, (cx, cy)

        return fill, stroke, extra, (0.0, 0.0)

    def boundingRect(self) -> QRectF:
        if self._asset is None:
            return QRectF(
                0, 0, ORPHAN_SIZE_M, ORPHAN_SIZE_M,
            )

        r = QRectF()
        if not self._path.isEmpty():
            r = self._path.boundingRect()
        if not self._extra_path.isEmpty():
            er = self._extra_path.boundingRect()
            r = r.united(er) if not r.isEmpty() else er

        if r.isEmpty():
            return QRectF()
        # Р—Р°РїР°СЃ: 0.3 Рј РѕР±С‹С‡РЅС‹Р№ + 2.5 Рј РїСЂРё РІС‹РґРµР»РµРЅРёРё (РґР»СЏ РїРѕРґРїРёСЃРµР№ СЂР°Р·РјРµСЂРѕРІ)
        pad = 2.5 if self.isSelected() else 0.3
        return r.adjusted(-pad, -pad, pad, pad)

    def shape(self) -> QPainterPath:
        """V16: РєР»РёРєР°Р±РµР»СЊРЅС‹Р№ bbox вЂ” С‚РµР»Рѕ РєРѕРјРїРѕРЅРµРЅС‚Р° Р·Р°С…РІР°С‚С‹РІР°РµС‚СЃСЏ
        РІ Р»СЋР±РѕР№ С‚РѕС‡РєРµ, Р° РЅРµ С‚РѕР»СЊРєРѕ РїРѕ РєРѕРЅС‚СѓСЂСѓ."""
        result = QPainterPath()
        if self._path.isEmpty() and self._extra_path.isEmpty():
            return result

        r = QRectF()
        if not self._path.isEmpty():
            r = self._path.boundingRect()
        if not self._extra_path.isEmpty():
            er = self._extra_path.boundingRect()
            r = r.united(er) if not r.isEmpty() else er
        if not r.isEmpty():
            result.addRect(r)
        return result

    def paint(self, painter, option, widget=None) -> None:
        painter.setRenderHint(painter.RenderHint.Antialiasing, True)

        # Orphan
        if self._asset is None:
            pen = QPen(ORPHAN_COLOR, 1.5)
            pen.setCosmetic(True)
            pen.setStyle(Qt.PenStyle.DashLine)
            painter.setPen(pen)
            painter.setBrush(Qt.BrushStyle.NoBrush)

    def _paint_dimensions(self, painter) -> None:
        """РџРѕРґРїРёСЃРё С€РёСЂРёРЅС‹/РІС‹СЃРѕС‚С‹ РІ СЌРєСЂР°РЅРЅС‹С… РїРёРєСЃРµР»СЏС… (С‚РѕР»СЊРєРѕ Сѓ РІС‹РґРµР»РµРЅРЅРѕРіРѕ)."""
        r = QRectF()
        if not self._path.isEmpty():
            r = self._path.boundingRect()
        if not self._extra_path.isEmpty():
            er = self._extra_path.boundingRect()
            r = r.united(er) if not r.isEmpty() else er
        if r.isEmpty():
            return

        w = r.width()
        h = r.height()

        scene = self.scene()
        if scene is None:
            return
        views = scene.views()
        if not views:
            return
        view = views[0]

        # РўРѕС‡РєРё РІ scene-РєРѕРѕСЂРґРёРЅР°С‚Р°С…
        bottom_scene = self.mapToScene(
            QPointF(r.center().x(), r.bottom())
        )
        right_scene = self.mapToScene(
            QPointF(r.right(), r.center().y())
        )

        # РџРµСЂРµРІРѕРґРёРј РІ РїРёРєСЃРµР»Рё viewport
        bottom_px = view.mapFromScene(bottom_scene)
        right_px = view.mapFromScene(right_scene)

        from PySide6.QtGui import QFont, QFontMetrics

        painter.save()
        painter.resetTransform()  # РѕС‚РєР»СЋС‡Р°РµРј РјР°СЃС€С‚Р°Р± СЃС†РµРЅС‹

        font = QFont("sans-serif")
        font.setPixelSize(12)
        painter.setFont(font)
        painter.setPen(QPen(QColor("#0055CC")))

        fm = QFontMetrics(font)

        # РЁРёСЂРёРЅР° вЂ” СЃРЅРёР·Сѓ РїРѕ С†РµРЅС‚СЂСѓ
        w_text = f"{w:.2f} Рј"
        w_tw = fm.horizontalAdvance(w_text)
        painter.drawText(
            QPointF(bottom_px.x() - w_tw / 2, bottom_px.y() + 16),
            w_text,
        )

        # Р’С‹СЃРѕС‚Р° вЂ” СЃРїСЂР°РІР°
        h_text = f"{h:.2f} Рј"
        painter.drawText(
            QPointF(right_px.x() + 8, right_px.y() + 4),
            h_text,
        )

        painter.restore()

    def apply_from_component(self) -> None:
        """РџРµСЂРµСЃРѕР±СЂР°С‚СЊ transform РёР· РєРѕРјРїРѕРЅРµРЅС‚Р° (РїРѕСЃР»Рµ РїСЂР°РІРєРё РІ РїР°РЅРµР»Рё)."""
        c = self._component
        self.setPos(c.x, c.y)
        self.setRotation(-c.rotation)
        self.setScale(c.scale)
        self.setZValue(self._compute_z())
        self.update()
        self.moved.emit()

    def wall_height_m(self) -> float:
        """V11: РІС‹СЃРѕС‚Р° РєРѕРЅС‚СѓСЂР° РІ РјРµС‚СЂР°С… (СѓР¶Рµ СЃ override).

        Р’РѕР·РІСЂР°С‰Р°РµС‚ 0.0, РµСЃР»Рё РєРѕРЅС‚СѓСЂ РїСѓСЃС‚.
        """
        if self._path.isEmpty():

    def slot_y_from_bottom(self, slot_key: str) -> float | None:
        """V11: СЂР°СЃСЃС‚РѕСЏРЅРёРµ СЃР»РѕС‚Р° РѕС‚ РЅРёР·Р° РєРѕРЅС‚СѓСЂР° (РІ РјРµС‚СЂР°С…).

        Р’РѕР·РІСЂР°С‰Р°РµС‚ None, РµСЃР»Рё СЃР»РѕС‚ РЅРµ РЅР°Р№РґРµРЅ.
        """
        if self._path.isEmpty():
            return None
        slots = self.slots_local()
        pos = slots.get(slot_key)
        if pos is None:
            return None
        rect = self._path.boundingRect()
        # rect.bottom() = max_y = foundation (РЅРёР· РєРѕРЅС‚СѓСЂР°)
        # pos[1] вЂ” y СЃР»РѕС‚Р° РІ С‚РµС… Р¶Рµ (path) РєРѕРѕСЂРґРёРЅР°С‚Р°С….
        return rect.bottom() - pos[1]

    def anchors_local(self) -> dict[str, tuple[float, float]]:
        """Anchor'С‹ СЃСЃС‹Р»РѕС‡РЅРѕРіРѕ asset'Р° РІ Р›РћРљРђР›Р¬РќР«РҐ РєРѕРѕСЂРґРёРЅР°С‚Р°С… item.

        V11b: СѓС‡РёС‚С‹РІР°РµС‚ param_overrides вЂ” anchor-С‚РѕС‡РєРё С‚РѕР¶Рµ СЃРґРІРёРіР°СЋС‚СЃСЏ,
        РєРѕРіРґР° РєРѕРЅС‚СѓСЂ СЂР°СЃС‚СЏРЅСѓС‚ С‡РµСЂРµР· РїР°СЂР°РјРµС‚СЂС‹.
        """
        if self._asset is None:
            return {}

        raw = self._asset.anchors()
        if not raw:
            return {}

        # РЎРѕР±СЂР°С‚СЊ СЃСѓРјРјР°СЂРЅСѓСЋ РґРµР»СЊС‚Сѓ РїРѕ СѓР·Р»Р°Рј РѕС‚ override-РїР°СЂР°РјРµС‚СЂРѕРІ
        overrides = getattr(self._component, "param_overrides", None) or {}
        deltas_per_node: dict[str, tuple[float, float]] = {}

        if overrides:
            params = getattr(self._asset, "parameters", None) or {}
            sg_all = getattr(self._asset, "semantic_groups", None) or {}
            for p in params.values():
                ov = overrides.get(p.name)
                if ov is None:
                    continue
                shift = float(ov) - float(p.value)
                if abs(shift) < 1e-12:
                    continue
                nd = compute_delta_for_parameter(p, shift, sg_all)
                for nid, (dx, dy) in nd.items():
                    px, py = deltas_per_node.get(nid, (0.0, 0.0))
                    deltas_per_node[nid] = (px + dx, py + dy)

        sg = getattr(self._asset, "semantic_groups", None) or {}

        result = {}
        for tag, (x, y) in raw.items():
            avg_dx = 0.0
            avg_dy = 0.0

            # V15: СЃРЅР°С‡Р°Р»Р° РёС‰РµРј РіСЂСѓРїРїСѓ anchor_<tag> (РЅР°РїСЂРёРјРµСЂ,
            # anchor_top), Рё С‚РѕР»СЊРєРѕ РµСЃР»Рё РµС‘ РЅРµС‚ вЂ” <tag> (top).
            # РўР°Рє СЏРєРѕСЂСЊ РЅРµ РµРґРµС‚ РІРјРµСЃС‚Рµ СЃ "С‡СѓР¶РёРјРё" СѓР·Р»Р°РјРё, РєРѕС‚РѕСЂС‹Рµ
            # РїРѕРїР°Р»Рё РІ РґСЂСѓРіСѓСЋ РіСЂСѓРїРїСѓ (right/left/top).
            group = None
            for g in sg.values():
                if g.name == f"anchor_{tag}":
                    group = g
                    break
            if group is None:
                for g in sg.values():
                    if g.name == tag:
                        group = g
                        break

            if group is not None and group.node_ids:
                dxs, dys = [], []
                for nid in group.node_ids:
                    d = deltas_per_node.get(nid)
                    if d is not None:
                        dxs.append(d[0])
                        dys.append(d[1])
                if dxs:
                    avg_dx = sum(dxs) / len(dxs)
                    avg_dy = sum(dys) / len(dys)

            result[tag] = (
                x + avg_dx - self._center_x,
                y + avg_dy - self._center_y,
            )

        return result

    def slots_local(self) -> dict[str, tuple[float, float]]:
        """V9d-4d: СЃР»РѕС‚С‹ (auto_rule) вЂ” РїРµСЂРµСЃС‡РёС‚С‹РІР°СЋС‚СЃСЏ РЅР° Р»РµС‚Сѓ
        СЃ СѓС‡С‘С‚РѕРј param_overrides.

        Р’РѕР·РІСЂР°С‰Р°РµС‚ {slot_key: (x, y)} РІ Р›РћРљРђР›Р¬РќР«РҐ РєРѕРѕСЂРґРёРЅР°С‚Р°С… item.
        РљР»СЋС‡ РІРёРґР° "slot_<groupname>_<idx>".
        """
        if self._asset is None:
            return {}

        sg = getattr(self._asset, "semantic_groups", None) or {}
        rule_groups = [
            g for g in sg.values()
            if getattr(g, "auto_rule", None) and g.node_ids
        ]
        if not rule_groups:
            return {}

        # V16: С‚РѕС‡РєРё РёР· Р’РЎР•РҐ РІРёРґРёРјС‹С… СЃР»РѕС‘РІ (РёРЅР°С‡Рµ СЃР»РѕС‚С‹
        # С‚РµСЂСЏСЋС‚СЃСЏ, РµСЃР»Рё РѕРЅРё РІ РЅРµР°РєС‚РёРІРЅРѕРј СЃР»РѕРµ).
        pts_by_id: dict[str, tuple[float, float]] = {}
        layers = getattr(self._asset, "layers", None) or []
        layer_geoms = []
        if layers:
            for layer in layers:
                if not getattr(layer, "visible", True):
                    continue
                layer_geoms.append(layer.geometry or {})
        if not layer_geoms:
            layer_geoms = [self._asset.geometry or {}]

        for g_geom in layer_geoms:
            contour = g_geom.get("contour", [])
            node_ids = g_geom.get("node_ids", [])
            for i, nid in enumerate(node_ids):
                if i < len(contour):
                    x, y = contour[i]
                    pts_by_id[nid] = (float(x), float(y))
            extra_pts = g_geom.get("extra_points", [])
            extra_ids = g_geom.get("extra_node_ids", [])
            for i, nid in enumerate(extra_ids):
                if i < len(extra_pts):
                    x, y = extra_pts[i]
                    pts_by_id[nid] = (float(x), float(y))

        # Override-СЃРґРІРёРіРё РїРѕ СѓР·Р»Р°Рј
        overrides = getattr(self._component, "param_overrides", None) or {}
        deltas: dict[str, tuple[float, float]] = {}
        if overrides:
            params = getattr(self._asset, "parameters", None) or {}
            for p in params.values():
                ov = overrides.get(p.name)
                if ov is None:
                    continue
                shift = float(ov) - float(p.value)
                if abs(shift) < 1e-12:
                    continue
                nd = compute_delta_for_parameter(p, shift, sg)
                for nid, (dx, dy) in nd.items():
                    px, py = deltas.get(nid, (0.0, 0.0))
                    deltas[nid] = (px + dx, py + dy)

        def _pos(nid):
            p = pts_by_id.get(nid)
            if p is None:
                return None
            dx, dy = deltas.get(nid, (0.0, 0.0))
            return (p[0] + dx, p[1] + dy)

        def _centroid(group):
            xs, ys = [], []
            for nid in group.node_ids:
                pp = _pos(nid)
                if pp is None:
                    continue
                xs.append(pp[0])
                ys.append(pp[1])
            if not xs:
                return None
            return (sum(xs) / len(xs), sum(ys) / len(ys))

        result: dict[str, tuple[float, float]] = {}

        for group in rule_groups:
            rule = group.auto_rule
            axis = rule.get("axis", "y")
            axis_idx = 0 if axis == "x" else 1
            step = float(rule.get("step", 1.0))
            if abs(step) < 1e-9:
                continue

            # V9c-РґРѕРї: РІСЃРµ РЅРµ-auto С‚РѕС‡РєРё РіСЂСѓРїРїС‹ вЂ” РЅРµР·Р°РІРёСЃРёРјС‹Рµ С€Р°Р±Р»РѕРЅС‹
            templates: list = []
            for nid in group.node_ids:
                if not nid.startswith("e_auto_"):
                    p = _pos(nid)
                    if p is not None:
                        templates.append(p)
            if not templates:
                continue

            # Р“СЂР°РЅРёС†Р° (until_group)
            until_name = rule.get("until_group") or ""
            limit_val = None
            if until_name:
                limit_group = None
                for gg in sg.values():
                    if gg.name == until_name:
                        limit_group = gg
                        break
                if limit_group is not None:
                    c = _centroid(limit_group)
                    if c is not None:
                        limit_val = c[axis_idx]

            max_count = int(rule.get("max_count", 30))

            # V15: РІС‚РѕСЂР°СЏ РѕСЃСЊ (СЃРµС‚РєР° С‚РѕС‡РµРє)
            axis2_name = rule.get("axis_x") or ""
            step2 = float(rule.get("step_x") or 0.0)
            until2_name = rule.get("until_group_x") or ""
            axis2_idx = None
            limit2_val = None
            if (axis2_name and abs(step2) > 1e-9 and until2_name):
                axis2_idx = 0 if axis2_name == "x" else 1
                if axis2_idx != axis_idx:
                    for gg in sg.values():
                        if gg.name == until2_name:
                            c = _centroid(gg)
                            if c is not None:
                                limit2_val = c[axis2_idx]
                            break
                    if limit2_val is None:
                        axis2_idx = None

            for tpl_idx, template_pos in enumerate(templates):
                # Р‘Р°Р·РѕРІР°СЏ С†РµРїРѕС‡РєР° РїРѕ РїРµСЂРІРѕР№ РѕСЃРё
                chain1 = [tuple(template_pos)]
                if limit_val is not None:
                    cur = template_pos[axis_idx] + step
                    i1 = 0
                    while i1 < max_count:
                        if step > 0 and cur >= limit_val:
                            break
                        if step < 0 and cur <= limit_val:
                            break
                        pt = list(template_pos)
                        pt[axis_idx] = cur
                        chain1.append(tuple(pt))
                        cur += step
                        i1 += 1

                for i1, base_pt in enumerate(chain1):
                    # Р‘Р°Р·РѕРІР°СЏ С‚РѕС‡РєР° вЂ” СЃС‚Р°СЂР°СЏ СЃС…РµРјР° РєР»СЋС‡Р°
                    key = f"slot_{group.id}_{tpl_idx}_{i1}"
                    result[key] = (
                        base_pt[0] - self._center_x,
                        base_pt[1] - self._center_y,
                    )

                    # Р’С‚РѕСЂР°СЏ РѕСЃСЊ (РєРѕРїРёРё РІР±РѕРє)
                    if axis2_idx is None or limit2_val is None:
                        continue
                    cur2 = base_pt[axis2_idx] + step2
                    j = 1
                    while j <= max_count:
                        if step2 > 0 and cur2 >= limit2_val:
                            break
                        if step2 < 0 and cur2 <= limit2_val:
                            break
                        pt2 = list(base_pt)
                        pt2[axis2_idx] = cur2
                        key2 = (
                            f"slot_{group.id}_{tpl_idx}_{i1}_x{j}"
                        )
                        result[key2] = (
                            pt2[0] - self._center_x,
                            pt2[1] - self._center_y,
                        )
                        cur2 += step2
                        j += 1

        return result

    def _try_snap(self) -> None:
        """РќР°Р№С‚Рё Р±Р»РёР¶Р°Р№С€РёР№ СЃРѕРІРјРµСЃС‚РёРјС‹Р№ anchor Рё РїСЂРёР»РёРїРЅСѓС‚СЊ."""
        scene = self.scene()
        if scene is None:
            return

        # V14: Рє РґСЂСѓРіРёРј РєРѕРјРїРѕРЅРµРЅС‚Р°Рј (anchor/slot) РїСЂРёР»РёРїР°СЋС‚ С‚РѕР»СЊРєРѕ
        # РѕРєРЅР°/РґРІРµСЂРё/РґРµРєРѕСЂ. РЎС‚РµРЅС‹, Р±Р°С€РЅРё, РєСЂС‹С€Рё вЂ” СЃР°РјРё Рє СЃРµР±Рµ
        # РЅРµ С†РµРїР»СЏСЋС‚СЃСЏ, РЅРѕ РјРѕРіСѓС‚ РїСЂРёР»РёРїРЅСѓС‚СЊ Рє РѕРїРѕСЂРЅРѕР№ Р»РёРЅРёРё.
        # V16: Рє РґСЂСѓРіРёРј РєРѕРјРїРѕРЅРµРЅС‚Р°Рј РїСЂРёР»РёРїР°СЋС‚ "РјРµР»РєРёРµ" СЌР»РµРјРµРЅС‚С‹.
        # РћРїСЂРµРґРµР»СЏРµРј РїРѕ С‚РёРїСѓ Р°СЃСЃРµС‚Р° РР›Р РїРѕ РёРјРµРЅРё РєРѕРјРїРѕРЅРµРЅС‚Р°
        # (РїРѕС‚РѕРјСѓ С‡С‚Рѕ РѕРєРЅР° С‡Р°СЃС‚Рѕ РёРјРµСЋС‚ С‚РёРї tower_body).
        ATTACHABLE = ("window", "door", "ornament")
        _mt = getattr(self._asset, "type", "") if self._asset else ""
        _name_lower = (
            getattr(self._component, "name", "") or ""
        ).lower()
        _name_ok = (
            "okno" in _name_lower
            or "РѕРєРЅРѕ" in _name_lower
            or "door" in _name_lower
            or "РґРІРµСЂ" in _name_lower
        )
        can_attach = (_mt in ATTACHABLE) or _name_ok

        my_world = self.anchors_world()
        if not my_world:
            self.clear_snap_highlight()
            self._component.clear_attachment()
            return

        views = scene.views()
        ppm = 50.0
        if views:
            ppm = abs(views[0].transform().m11()) or 50.0
        threshold_m = SNAP_THRESHOLD_PX / ppm

        best = None  # (dist, my_tag, their_item, their_tag, dx, dy)

        # V15: anchor-РїРѕРёСЃРє РёРґС‘С‚ Р’РЎР•Р“Р”Рђ, РЅРѕ РґР»СЏ РЅРµ-attachable
        # РґРѕРїСѓСЃРєР°РµРј С‚РѕР»СЊРєРѕ РїР°СЂСѓ (bottom в†’ top) вЂ” РєСЂС‹С€Р°/РєСѓРїРѕР»/Р±Р°С€РЅСЏ
        # СЃС‚Р°РІСЏС‚СЃСЏ СЃРІРµСЂС…Сѓ. РћСЃС‚Р°Р»СЊРЅС‹Рµ РїР°СЂС‹ РёРј Р·Р°РїСЂРµС‰РµРЅС‹.
        if True:
          for other in scene.items():
            if other is self:
                continue
            if not isinstance(other, ComponentItem):
                continue
            their_world = other.anchors_world()
            if not their_world:
                continue

            # РџСЂРѕРїСѓСЃС‚РёС‚СЊ, РµСЃР»Рё СЌС‚РѕС‚ other СѓР¶Рµ РїСЂРёРІСЏР·Р°РЅ Рє РќРђРњ (РёРЅР°С‡Рµ С†РёРєР»)
            other_comp_check = getattr(other, "_component", None)
            if (other_comp_check is not None
                    and other_comp_check.attach_to
                    == self._component.id):
                continue

            # V9d-4: РѕРєРЅРѕ РќР• РїСЂРёРІСЏР·С‹РІР°РµРј Рє РґСЂСѓРіРѕРјСѓ РѕРєРЅСѓ РїРѕ anchor.
            # РћРєРЅР° С†РµРїР»СЏСЋС‚СЃСЏ С‚РѕР»СЊРєРѕ Рє СЃР»РѕС‚Р°Рј СЃС‚РµРЅ (РёР»Рё Рє РЅРµ-РѕРєРЅР°Рј).
            my_type = getattr(self._asset, "type", "") if self._asset else ""
            their_type = (
                getattr(other._asset, "type", "")
                if other._asset else ""
            )
            if my_type == "window" and their_type == "window":
                continue

            for my_tag, (mx, my) in my_world.items():
                for their_tag, (tx, ty) in their_world.items():
                    if (my_tag, their_tag) not in SNAP_PAIRS:
                        continue
                    # V15: "РІСЃС‚Р°С‚СЊ СЃРІРµСЂС…Сѓ" (bottomв†’top) СЂР°Р·СЂРµС€РµРЅРѕ
                    # РІСЃРµРј вЂ” РєСЂС‹С€Р°/РєСѓРїРѕР»/Р±Р°С€РЅСЏ СЃС‚Р°РІСЏС‚СЃСЏ РЅР° СЃС‚РµРЅСѓ.
                    # Р”СЂСѓРіРёРµ РїР°СЂС‹ (topв†’bottom) вЂ” С‚РѕР»СЊРєРѕ attachable.
                    if (not can_attach
                            and (my_tag, their_tag) != ("bottom", "top")):
                        continue

                    # V20: РЅРµ-attachable (Р±Р°С€РЅСЏ/СЃС‚РµРЅР°/РєСЂС‹С€Р°) РЅРµ Р»РёРїРЅРµС‚
                    # Рє РѕРєРЅР°Рј/РґРІРµСЂСЏРј/РґРµРєРѕСЂСѓ вЂ” С‚РѕР»СЊРєРѕ Рє РєСЂСѓРїРЅРѕРјСѓ
                    # (СЃС‚РµРЅР°, Р±Р°С€РЅСЏ, РєСЂС‹С€Р°, РєСѓРїРѕР»).
                    if not can_attach:
                        _o_asset = getattr(other, "_asset", None)
                        _o_type = (
                            getattr(_o_asset, "type", "")
                            if _o_asset is not None else ""
                        )
                        if _o_type in ATTACHABLE:
                            continue

                    dx = tx - mx
                    dy = ty - my
                    dist = (dx * dx + dy * dy) ** 0.5
                    if dist <= threshold_m:
                        if best is None or dist < best[0]:
                            best = (dist, my_tag, other, their_tag,
                                    dx, dy)

        # V9d-4: РїРѕРёСЃРє СЃР»РѕС‚РѕРІ (auto_rule) РЅР° СЃРѕСЃРµРґРЅРёС… СЃС‚РµРЅР°С….
        # РџСЂРёРѕСЂРёС‚РµС‚ Сѓ СЃР»РѕС‚Р° вЂ” РµСЃР»Рё РѕРЅ Р±Р»РёР¶Рµ, С‡РµРј anchor-РїР°СЂР°.
        best_slot = None
        my_bottom = my_world.get("bottom")
        if my_bottom is not None:
            mx, my = my_bottom

            # РЎРѕР±СЂР°С‚СЊ Р·Р°РЅСЏС‚С‹Рµ СЃР»РѕС‚С‹: {parent_comp_id: {slot_key, ...}}
            occupied: dict[str, set] = {}
            for cand in scene.items():
                if cand is self:
                    continue
                if not isinstance(cand, ComponentItem):
                    continue
                c_comp = getattr(cand, "_component", None)
                if c_comp is None or not c_comp.attach_to:
                    continue
                if c_comp.parent_anchor.startswith("slot_"):
                    occupied.setdefault(
                        c_comp.attach_to, set()
                    ).add(c_comp.parent_anchor)

            if not can_attach:
                pass
            else:
              for other in scene.items():
                if other is self:
                    continue
                if not isinstance(other, ComponentItem):
                    continue
                their_slots = other.slots_world()
                if not their_slots:
                    continue

                o_comp = getattr(other, "_component", None)
                if o_comp is None:
                    continue
                if o_comp.id == self._component.id:
                    continue
                # РџСЂРѕРїСѓСЃС‚РёС‚СЊ, РµСЃР»Рё other РїСЂРёРІСЏР·Р°РЅ Рє РќРђРњ
                if o_comp.attach_to == self._component.id:
                    continue

                taken = occupied.get(o_comp.id, set())

                for slot_key, (sx, sy) in their_slots.items():
                    if slot_key in taken:
                        continue
                    dx = sx - mx
                    dy = sy - my
                    dist = (dx * dx + dy * dy) ** 0.5
                    if dist <= threshold_m:
                        if best_slot is None or dist < best_slot[0]:
                            best_slot = (
                                dist, "bottom", other, slot_key,
                                dx, dy,
                            )

        # Р’С‹Р±РѕСЂ: РµСЃР»Рё СЃР»РѕС‚ Р±Р»РёР¶Рµ anchor-РїР°СЂС‹ вЂ” Р±РµСЂС‘Рј СЃР»РѕС‚
        if best_slot is not None:
            if best is None or best_slot[0] < best[0]:
                best = best_slot

        # V13: snap Рє РѕРїРѕСЂРЅРѕР№ Р»РёРЅРёРё (ground line).
        # Р•СЃР»Рё РјРѕР№ bottom anchor Р±Р»РёР·РєРѕ Рє Y Р»РёРЅРёРё вЂ” snap РїРѕ Y.
        scene = self.scene()
        if scene is not None and hasattr(scene, "ground_line_y"):
            gy = scene.ground_line_y()
            if gy is not None and scene.ground_line_visible():
                my_bottom = my_world.get("bottom")
                if my_bottom is not None:
                    my_y = my_bottom[1]
                    dy = gy - my_y
                    if abs(dy) <= threshold_m:
                        # Р•СЃР»Рё Р»РёРЅРёСЏ Р±Р»РёР¶Рµ, С‡РµРј С‚РµРєСѓС‰РёР№ best
                        if best is None or abs(dy) < best[0]:
                            best = (
                                abs(dy), "__ground__",
                                None, "__ground__",
                                0.0, dy,
                            )

        # V13: СЃР±СЂРѕСЃ С„Р»Р°РіР° вЂ” СѓСЃС‚Р°РЅРѕРІРёС‚СЃСЏ С‚РѕР»СЊРєРѕ РµСЃР»Рё СЂРµР°Р»СЊРЅРѕ snap Рє ground
        self._snap_to_ground = False

        if best is None:
            self.clear_snap_highlight()
            return

        _, my_tag, other, their_tag, dx, dy = best

        # РЎРЅР°С‡Р°Р»Р° РѕС‡РёС‰Р°РµРј СЃС‚Р°СЂСѓСЋ РїР°СЂСѓ
        self.clear_snap_highlight()

        # V13: snap Рє ground line вЂ” СЃРґРІРёРіР°РµРј С‚РѕР»СЊРєРѕ РїРѕ Y,
        # attachment РЅРµ СЃРѕР·РґР°С‘Рј (СЌС‚Рѕ РЅРµ РєРѕРјРїРѕРЅРµРЅС‚).
        if other is None:
            self.setPos(self.pos().x(), self.pos().y() + dy)
            self._component.x = self.pos().x()
            self._component.y = self.pos().y()
            self._snap_to_ground = True
            return

        # РЎРґРІРёРіР°РµРј СЃРµР±СЏ РЅР° (dx, dy), С‡С‚РѕР±С‹ РЅР°С€ anchor СЃРѕРІРїР°Р» СЃ РёС…
        self.setPos(self.pos().x() + dx, self.pos().y() + dy)

        # РџРѕРґСЃРІРµС‚РєР°
        self.set_highlighted_anchor(my_tag)
        other.set_highlighted_anchor(their_tag)

        # V11c: СЃРѕС…СЂР°РЅРёС‚СЊ РїСЂРёРІСЏР·РєСѓ РІ РјРѕРґРµР»СЊ
        other_comp = getattr(other, "_component", None)
        if other_comp is not None:
            already = (
                self._component.attach_to == other_comp.id
                and self._component.attach_anchor == my_tag
                and self._component.parent_anchor == their_tag
            )
            if not already:
                self._component.set_attachment(
                    other_comp.id, my_tag, their_tag,
                )
                print(f"[SNAP] attach_to={other_comp.id} "
                      f"my={my_tag} their={their_tag}")
        self._snap_partner = other

    def mouseDoubleClickEvent(self, event) -> None:
        """Р”РІРѕР№РЅРѕР№ РєР»РёРє вЂ” РІРѕР№С‚Рё РІ composite (РµСЃР»Рё СЌС‚Рѕ composite)."""
        if self._asset is not None:
            comps = getattr(self._asset, "components", None)

        super().mouseDoubleClickEvent(event)

    def mousePressEvent(self, event) -> None:
        if event.button() == Qt.MouseButton.LeftButton:
            self._drag_old_pos = (self.pos().x(), self.pos().y())
        super().mousePressEvent(event)

    def mouseMoveEvent(self, event) -> None:
        super().mouseMoveEvent(event)
        self._try_snap()

    def mouseReleaseEvent(self, event) -> None:
        super().mouseReleaseEvent(event)

        # Р•СЃР»Рё РєРѕРјРїРѕРЅРµРЅС‚ РїСЂРёРІСЏР·Р°РЅ Рє СЂРѕРґРёС‚РµР»СЋ С‡РµСЂРµР· anchor вЂ”
        # РїРѕР·РёС†РёРµР№ СѓРїСЂР°РІР»СЏРµС‚ reflow, РЅРµ С‚СЂРѕРіР°РµРј РµС‘ СЂСѓРєР°РјРё.
        # РРЅР°С‡Рµ Р»СЋР±РѕР№ РєР»РёРє СЃРґРІРёРіР°Р» Р±С‹ С‚РѕС‡РєСѓ СЃС‚С‹РєРѕРІРєРё.
        attached = bool(getattr(self._component, "attach_to", ""))

        if attached:
            pos = self.pos()
            sx = pos.x()
            sy = pos.y()
        elif self._snap_partner is not None:
            # РЎСЂР°Р±РѕС‚Р°Р» snap вЂ” РЅРµ РѕРєСЂСѓРіР»СЏРµРј, С‚РѕС‡РЅР°СЏ РїРѕР·РёС†РёСЏ
            pos = self.pos()
            sx = pos.x()
            sy = pos.y()
        else:
            # РЎРІРѕР±РѕРґРЅС‹Р№ РєРѕРјРїРѕРЅРµРЅС‚ вЂ” СЃРЅР°РїРёРј Рє СЃРµС‚РєРµ 0.1 Рј
            pos = self.pos()
            step = self._snap_step
            sx = round(pos.x() / step) * step
            sy = round(pos.y() / step) * step

            if abs(sx - pos.x()) > 1e-9 or abs(sy - pos.y()) > 1e-9:
                self.setPos(sx, sy)

        # Р—Р°РїРёСЃС‹РІР°РµРј РІ РєРѕРјРїРѕРЅРµРЅС‚
        self._component.x = sx
        self._component.y = sy
        self.moved.emit()

        # Undo: РµСЃР»Рё РїРѕР·РёС†РёСЏ РёР·РјРµРЅРёР»Р°СЃСЊ вЂ” СЃРёРіРЅР°Р» РґР»СЏ РєРѕРјР°РЅРґС‹
        if self._drag_old_pos is not None:
            ox, oy = self._drag_old_pos
            if abs(ox - sx) > 1e-6 or abs(oy - sy) > 1e-6:
                self.drag_finished.emit(
                    self._component.id, ox, oy, sx, sy,
                )
            self._drag_old_pos = None

        # РЈР±РёСЂР°РµРј РїРѕРґСЃРІРµС‚РєСѓ snap
        self._snap_to_ground = False
        self.clear_snap_highlight()

    def itemChange(self, change, value):
        if change == QGraphicsItem.GraphicsItemChange.ItemSelectedHasChanged:
            self.prepareGeometryChange()
            self.update()
            from PySide6.QtCore import QTimer
            QTimer.singleShot(0, self._rebuild_handles)
        elif change == QGraphicsItem.GraphicsItemChange.ItemPositionHasChanged:
            # V12: РїСЂРё drag вЂ” СЌРјРёС‚РёРј moved РЅР° РєР°Р¶РґРѕРµ РёР·РјРµРЅРµРЅРёРµ РїРѕР·РёС†РёРё,
            # С‡С‚РѕР±С‹ РґРµС‚Рё (РѕРєРЅР° РЅР° СЃР»РѕС‚Р°С…) РґРІРёРіР°Р»РёСЃСЊ СЃРёРЅС…СЂРѕРЅРЅРѕ.
            if self._drag_old_pos is not None:
                self.moved.emit()
        return super().itemChange(change, value)

    def _get_drop_position(self) -> tuple:
        """РџРѕР·РёС†РёСЏ РґР»СЏ РЅРѕРІРѕРіРѕ РєРѕРјРїРѕРЅРµРЅС‚Р° вЂ” РїРѕРґ РјС‹С€СЊСЋ РёР»Рё РІ С†РµРЅС‚СЂРµ РІРёРґР°."""
        try:
            from PySide6.QtGui import QCursor
            global_pos = QCursor.pos()
            local_pos = self._canvas.viewport().mapFromGlobal(global_pos)
            if self._canvas.viewport().rect().contains(local_pos):
                scene_pos = self._canvas.mapToScene(local_pos)
                return (scene_pos.x(), scene_pos.y())
            center = self._canvas.viewport().rect().center()
            scene_pos = self._canvas.mapToScene(center)
            return (scene_pos.x(), scene_pos.y())
        except Exception:
            return (0.0, 0.0)

    def _on_add_asset(self, asset_id: str) -> None:
        """Р”РІРѕР№РЅРѕР№ РєР»РёРє РІ РїР°Р»РёС‚СЂРµ вЂ” РґРѕР±Р°РІРёС‚СЊ РєРѕРјРїРѕРЅРµРЅС‚ РЅР° СЃС†РµРЅСѓ."""
        # Р•СЃР»Рё РІ preview вЂ” РІС‹Р№С‚Рё
        if self._preview_mode:
            self._exit_preview()

        if self._registry is None:
            self.statusBar().showMessage("Р РµРµСЃС‚СЂ Р°СЃСЃРµС‚РѕРІ РЅРµРґРѕСЃС‚СѓРїРµРЅ", 3000)
            return

        asset = self._registry.get(asset_id)
        if asset is None:
            self.statusBar().showMessage(
                f"РђСЃСЃРµС‚ {asset_id} РЅРµ РЅР°Р№РґРµРЅ", 3000)
            return

        # РџРѕР·РёС†РёСЏ вЂ” РїРѕРґ РєСѓСЂСЃРѕСЂРѕРј РјС‹С€Рё РёР»Рё РІ С†РµРЅС‚СЂРµ РІРёРґР°
        drop_x, drop_y = self._get_drop_position()

        comp = Component(
            asset_id=asset_id,
            x=drop_x,
            y=drop_y,
            rotation=0.0,
            scale=1.0,
            name=asset.name,
        )

        # Р§РµСЂРµР· undo-РєРѕРјР°РЅРґСѓ
        cmd = AddComponentCommand(
            composite=self._current_composite,
            comp=comp,
            on_create_item=self._create_component_item,
            on_remove_item=self._remove_component_item,
        )
        self._undo_stack.push(cmd)

        self.statusBar().showMessage(
            f"Р”РѕР±Р°РІР»РµРЅ: {asset.name} (id={comp.id[:8]}). "
            f"Р’СЃРµРіРѕ: {self._current_composite.component_count()}",
            4000,
        )

    def _on_save(self) -> None:
        if self._current_asset_id is None:
            self._on_save_as()
            return

        # РћР±РЅРѕРІР»СЏРµРј РёРјСЏ Рё СЃРѕС…СЂР°РЅСЏРµРј
        self._current_composite.name = self._current_asset_name

        # V13: СЃРѕС…СЂР°РЅРёС‚СЊ ground line РёР· СЃС†РµРЅС‹
        if self._scene is not None and hasattr(
            self._scene, "ground_line_y"
        ):
            self._current_composite.ground_line_y = (
                self._scene.ground_line_y()
            )
            self._current_composite.ground_line_visible = (
                self._scene.ground_line_visible()
            )

        try:
            save_asset(self._current_composite)
        except StorageError as e:
            QMessageBox.critical(
                self, "РћС€РёР±РєР° СЃРѕС…СЂР°РЅРµРЅРёСЏ",
                f"РќРµ СѓРґР°Р»РѕСЃСЊ СЃРѕС…СЂР°РЅРёС‚СЊ:\n\n{e}",
            )
            return

        if self._registry is not None:
            self._registry.load_all()
        self._load_assets()

        self.statusBar().showMessage(
            f"РЎРѕС…СЂР°РЅРµРЅРѕ: {self._current_asset_name}", 3000,
        )

    def _on_save_as(self) -> None:
        name, ok = QInputDialog.getText(
            self, "РЎРѕС…СЂР°РЅРёС‚СЊ РєР°Рє",
            "РРјСЏ РєРѕРјРїРѕР·РёС‚РЅРѕРіРѕ Р°СЃСЃРµС‚Р°:",
            text=self._current_asset_name,
        )
        if not ok:
            return
        name = name.strip()
        if not name:
            return

        # РќРѕРІС‹Р№ Asset вЂ” СЃ РЅРѕРІС‹Рј uuid, РєРѕРїРёСЂСѓРµРј СЃРѕРґРµСЂР¶РёРјРѕРµ
        new_asset = Asset(name=name, type_="tower_body")
        new_asset.components = dict(self._current_composite.components)

        # V13: СЃРєРѕРїРёСЂРѕРІР°С‚СЊ ground line
        if self._scene is not None and hasattr(
            self._scene, "ground_line_y"
        ):
            new_asset.ground_line_y = self._scene.ground_line_y()
            new_asset.ground_line_visible = (
                self._scene.ground_line_visible()
            )

        try:
            save_asset(new_asset)
        except StorageError as e:
            QMessageBox.critical(
                self, "РћС€РёР±РєР° СЃРѕС…СЂР°РЅРµРЅРёСЏ",
                f"РќРµ СѓРґР°Р»РѕСЃСЊ СЃРѕС…СЂР°РЅРёС‚СЊ:\n\n{e}",
            )
            return

        self._current_composite = new_asset
        self._current_asset_id = new_asset.id
        self._current_asset_name = name
        self._update_title()

        # РћР±РЅРѕРІРёС‚СЊ СЂРµРµСЃС‚СЂ Рё РїР°Р»РёС‚СЂСѓ вЂ” РЅРѕРІС‹Р№ Р°СЃСЃРµС‚
        if self._registry is not None:
            self._registry.load_all()
        self._load_assets()

        self.statusBar().showMessage(
            f"РЎРѕР·РґР°РЅ: {name} (id={new_asset.id[:8]})", 3000,
        )

    def _on_open(self) -> None:
        if self._preview_mode:
            self._preview_mode = False
            self._edit_backup = None

        # Р—Р°РіСЂСѓР¶Р°РµРј РІСЃРµ Р°СЃСЃРµС‚С‹, С„РёР»СЊС‚СЂСѓРµРј РєРѕРјРїРѕР·РёС‚РЅС‹Рµ
        try:
            all_assets = list_assets()
        except Exception as e:
            QMessageBox.warning(

            QMessageBox.information(
                self, "РќРµС‚ РєРѕРјРїРѕР·РёС‚РЅС‹С… Р°СЃСЃРµС‚РѕРІ",
                "РЎРЅР°С‡Р°Р»Р° СЃРѕР·РґР°Р№ Рё СЃРѕС…СЂР°РЅРё Р·Р°РјРѕРє С‡РµСЂРµР· "

            )
            return

        dlg = AssetOpenDialog(composites, mode="open", parent=self)
        if dlg.exec() != AssetOpenDialog.DialogCode.Accepted:
            # РњРѕР¶РµС‚ Р±С‹С‚СЊ СѓРґР°Р»РµРЅРёРµ
            del_id = dlg.deleted_id()
            if del_id:

        del_id = dlg.deleted_id()
        if del_id:
            self._do_delete_asset(del_id)
            return

        aid = dlg.selected_id()
        if not aid:
            return

        # Р—Р°РіСЂСѓР¶Р°РµРј
        target = next(
            (a for a in all_assets if a.id == aid), None
        )
        if target is None:
            return

        # РћС‚РєСЂС‹С‚РёРµ СЃ РІРµСЂС…РЅРµРіРѕ СѓСЂРѕРІРЅСЏ вЂ” СЃС‚РµРє РїСѓСЃС‚
        self._nav_stack.clear()

        # РџРµСЂРµС‡РёС‚Р°С‚СЊ СЂРµРµСЃС‚СЂ СЃ РґРёСЃРєР° вЂ” С‡С‚РѕР±С‹ РїРѕРґС…РІР°С‚РёС‚СЊ СЃРІРµР¶РёРµ
        # РїСЂР°РІРєРё РёР· Vector Editor (РіСЂСѓРїРїС‹, anchors, РєРѕРЅС‚СѓСЂС‹)
        if self._registry is not None:
            self._registry.load_all()

        self._load_composite(target)

    def _on_rename_asset(self, asset_id: str, new_name: str) -> None:

            )

        if self._registry is not None:
            self._registry.load_all()
        self._load_assets()

        self.statusBar().showMessage(
            "РџРµСЂРµРёРјРµРЅРѕРІР°РЅ: " + new_name, 3000,
        )

    def _on_delete_assets(self, asset_ids: list) -> None:
        """РЈРґР°Р»РёС‚СЊ СЃРїРёСЃРѕРє Р°СЃСЃРµС‚РѕРІ (РёР· РїР°Р»РёС‚СЂС‹)."""
        if not asset_ids:
            return

        names = []
        try:
            all_assets = list_assets()
            by_id = {a.id: a.name for a in all_assets}
            for aid in asset_ids[:3]:
                names.append(by_id.get(aid, aid[:8]))
        except Exception:
            names = [aid[:8] for aid in asset_ids[:3]]

        if len(asset_ids) == 1:
            msg = "РЈРґР°Р»РёС‚СЊ " + repr(names[0]) + "?"
        else:
            more = "" if len(asset_ids) <= 3 else " Рё РµС‰С‘ " + str(len(asset_ids) - 3)
            msg = (
                "РЈРґР°Р»РёС‚СЊ " + str(len(asset_ids)) + " Р°СЃСЃРµС‚РѕРІ?\n"
                "(" + ", ".join(names) + more + ")\n\n"
                "Р¤Р°Р№Р»С‹ Р±СѓРґСѓС‚ СѓРґР°Р»РµРЅС‹ СЃ РґРёСЃРєР°."
            )

        reply = QMessageBox.question(
            self, "РЈРґР°Р»РёС‚СЊ Р°СЃСЃРµС‚С‹", msg,
            QMessageBox.StandardButton.Yes
            | QMessageBox.StandardButton.No,
            QMessageBox.StandardButton.No,
        )
        if reply != QMessageBox.StandardButton.Yes:
            return

        deleted = 0
        for aid in asset_ids:
            try:
                delete_asset(aid)
                deleted += 1
            except Exception as e:
                print("[DELETE] РѕС€РёР±РєР° " + aid + ": " + str(e))

        if self._current_asset_id in asset_ids:
            self._reset_scene()

        if self._registry is not None:
            self._registry.load_all()
        self._load_assets()

        self.statusBar().showMessage(
            "РЈРґР°Р»РµРЅРѕ: " + str(deleted) + " РёР· " + str(len(asset_ids)),
            3000,
        )

                "РќРµС‚ СЃРѕС…СЂР°РЅС‘РЅРЅС‹С… РєРѕРјРїРѕР·РёС‚РЅС‹С… Р°СЃСЃРµС‚РѕРІ.",

        dlg = AssetOpenDialog(composites, mode="delete", parent=self)

    def _do_delete_asset(self, asset_id: str) -> None:
        """РЈРґР°Р»РёС‚СЊ Р°СЃСЃРµС‚ РїРѕ id + РѕР±РЅРѕРІРёС‚СЊ UI."""
        try:
            delete_asset(asset_id)
        except Exception as e:
            QMessageBox.warning(
                self, "РћС€РёР±РєР° СѓРґР°Р»РµРЅРёСЏ",
                f"РќРµ СѓРґР°Р»РѕСЃСЊ СѓРґР°Р»РёС‚СЊ:\n\n{e}",
            )
            return

        if self._current_asset_id == asset_id:
            self._reset_scene()

        if self._registry is not None:
            self._registry.load_all()
        self._load_assets()

        self.statusBar().showMessage(
            f"РђСЃСЃРµС‚ СѓРґР°Р»С‘РЅ (id={asset_id[:8]})", 3000,
        )

    def _load_composite(self, asset: Asset) -> None:
        """Р—Р°РіСЂСѓР·РёС‚СЊ РєРѕРјРїРѕР·РёС‚РЅС‹Р№ Asset РІ СЃС†РµРЅСѓ."""
        self._reset_scene()

        self._current_composite = asset
        self._current_asset_id = asset.id
        self._current_asset_name = asset.name
        self._update_title()

        # N РѕС‚РґРµР»СЊРЅС‹С… items вЂ” РїРѕ РѕРґРЅРѕРјСѓ РЅР° РєРѕРјРїРѕРЅРµРЅС‚ РІРµСЂС…РЅРµРіРѕ СѓСЂРѕРІРЅСЏ.
        # Р’Р»РѕР¶РµРЅРЅС‹Рµ composite РІРЅСѓС‚СЂРё РєР°Р¶РґРѕРіРѕ item РѕС‚СЂРёСЃСѓСЋС‚СЃСЏ СЂРµРєСѓСЂСЃРёРІРЅРѕ.
        n_orphan = 0
        for comp in asset.components.values():
            ref_asset = None
            if self._registry is not None:
                ref_asset = self._registry.get(comp.asset_id)
            if ref_asset is None:
                n_orphan += 1

            item = ComponentItem(
                comp, asset=ref_asset, registry=self._registry,
            )
            item.moved.connect(self._on_item_moved)
            item.enter_requested.connect(self._on_enter_composite)
            item.drag_finished.connect(self._on_item_drag_finished)
            item.param_changed.connect(self._on_prop_param_override)
            self._items_by_comp_id[comp.id] = item
            self._scene.addItem(item)

        msg = (
            f"Р—Р°РіСЂСѓР¶РµРЅРѕ: {asset.name} В· "
            f"{len(asset.components)} РєРѕРјРїРѕРЅРµРЅС‚РѕРІ"
        )
        if n_orphan:
            msg += f" В· {n_orphan} Р±РёС‚С‹С…"
        self.statusBar().showMessage(msg, 5000)

        # V13: Р·Р°РіСЂСѓР·РёС‚СЊ РѕРїРѕСЂРЅСѓСЋ Р»РёРЅРёСЋ РёР· РєРѕРјРїРѕР·РёС‚Р°
        if self._scene is not None:
            g_y = getattr(asset, "ground_line_y", None)
            g_v = bool(getattr(asset, "ground_line_visible", True))
            self._scene.set_ground_line(g_y, visible=g_v)
            if g_y is not None:
                self._ground_spin.blockSignals(True)
                self._ground_spin.setValue(float(g_y))
                self._ground_spin.blockSignals(False)
                self._act_ground.blockSignals(True)
                self._act_ground.setChecked(g_v)
                self._act_ground.blockSignals(False)
            else:
                self._act_ground.blockSignals(True)
                self._act_ground.setChecked(False)
                self._act_ground.blockSignals(False)

        # РџСЂРёРјРµРЅРёС‚СЊ РїСЂР°РІРёР»Р° СЃСЂР°Р·Сѓ РїСЂРё Р·Р°РіСЂСѓР·РєРµ
        self._apply_visibility_rules()
        self._apply_slot_visibility()

        # V-A: Р°РІС‚Рѕ-zoom РїРѕРґ РІСЃРµ РєРѕРјРїРѕРЅРµРЅС‚С‹ РєРѕРјРїРѕР·РёС‚Р°.
        from PySide6.QtCore import QTimer as _QT
        _QT.singleShot(0, self._fit_to_scene)

    def _fit_to_scene(self) -> None:
        """V-A: РїРѕРґРѕРіРЅР°С‚СЊ РІРёРґ canvas РїРѕРґ bbox РІСЃРµС… РєРѕРјРїРѕРЅРµРЅС‚РѕРІ."""
        if self._scene is None:
            return
        r = self._scene.itemsBoundingRect()
        if r.isEmpty():
            return
        # РњРёРЅРёРјСѓРј 1Г—1 Рј вЂ” С‡С‚РѕР±С‹ РЅРµ Р·СѓРјРёС‚СЊСЃСЏ РґРѕ Р±РµСЃРєРѕРЅРµС‡РЅРѕСЃС‚Рё
        # РЅР° РїРѕС‡С‚Рё РїСѓСЃС‚РѕР№ СЃС†РµРЅРµ.
        w = max(r.width(), 1.0)
        h = max(r.height(), 1.0)
        cx = r.center().x()
        cy = r.center().y()
        x0 = cx - w / 2
        x1 = cx + w / 2
        y0 = cy - h / 2
        y1 = cy + h / 2
        self._canvas.fit_to_rect(x0, y0, x1, y1)

    def _on_enter_composite(self, asset_id: str) -> None:
        """Р”РІРѕР№РЅРѕР№ РєР»РёРє РїРѕ composite вЂ” РІРѕР№С‚Рё РІРЅСѓС‚СЂСЊ."""
        if self._current_asset_id is None:
            reply = QMessageBox.question(
                self, "РЎРѕС…СЂР°РЅРёС‚СЊ С‚РµРєСѓС‰РёР№?",
                "Р§С‚РѕР±С‹ РІРѕР№С‚Рё РІРЅСѓС‚СЂСЊ, РЅР°РґРѕ СЃРѕС…СЂР°РЅРёС‚СЊ С‚РµРєСѓС‰РёР№ composite.\n\n"
                "РЎРѕС…СЂР°РЅРёС‚СЊ СЃРµР№С‡Р°СЃ?",
                QMessageBox.StandardButton.Yes
                | QMessageBox.StandardButton.No,
                QMessageBox.StandardButton.Yes,
            )
            if reply != QMessageBox.StandardButton.Yes:
                return
            self._on_save_as()
            if self._current_asset_id is None:
                return

        # Р—Р°РіСЂСѓР¶Р°РµРј sub-asset
        if self._registry is None:
            return
        sub = self._registry.get(asset_id)
        if sub is None:
            QMessageBox.warning(
                self, "РќРµ РЅР°Р№РґРµРЅРѕ",
                f"РђСЃСЃРµС‚ {asset_id} РЅРµ РЅР°Р№РґРµРЅ РІ Р±РёР±Р»РёРѕС‚РµРєРµ.",
            )
            return

        # РџСѓС€РёРј С‚РµРєСѓС‰РёР№ РІ СЃС‚РµРє
        self._nav_stack.append(
            (self._current_asset_id, self._current_asset_name)
        )

        # Р—Р°РіСЂСѓР¶Р°РµРј sub РєР°Рє РґРѕРєСѓРјРµРЅС‚
        self._load_composite(sub)

    def _on_nav_back(self) -> None:
        """Р’РµСЂРЅСѓС‚СЊСЃСЏ Рє СЂРѕРґРёС‚РµР»СЋ."""
        if not self._nav_stack:
            return

        parent_id, parent_name = self._nav_stack.pop()

        # РџРµСЂРµС‡РёС‚С‹РІР°РµРј parent СЃ РґРёСЃРєР°

            self._registry.load_all()

        else:
            parent = None

        if parent is None:
            QMessageBox.warning(
                self, "РќРµ РЅР°Р№РґРµРЅРѕ",
                f"Р РѕРґРёС‚РµР»СЊ {parent_id} РЅРµ РЅР°Р№РґРµРЅ.",
            )
            return

        self._load_composite(parent)

    def _apply_property(self, comp_id: str) -> None:
        """РџРѕСЃР»Рµ undo/redo СЃРІРѕР№СЃС‚РІР° вЂ” РѕР±РЅРѕРІРёС‚СЊ item + РїР°РЅРµР»СЊ."""
        comp = self._current_composite.get_component(comp_id)
        item = self._items_by_comp_id.get(comp_id)
        if item is not None:
            item.apply_from_component()

        if (self._properties._current_comp is not None
                and self._properties._current_comp.id == comp_id):
            self._properties.update_values(
                comp.x, comp.y, comp.rotation, comp.scale,
            )
            if hasattr(self._properties, "_layer_spin"):
                self._properties._muted = True
                self._properties._layer_spin.setValue(
                    int(getattr(comp, "layer", 0))
                )
                self._properties._muted = False

    def _on_prop_value_changed(
        self, comp_id: str, field: str, value: float,
    ) -> None:
        """РџР°РЅРµР»СЊ РёР·РјРµРЅРёР»Р° СЃРІРѕР№СЃС‚РІРѕ вЂ” С‡РµСЂРµР· undo-РєРѕРјР°РЅРґСѓ."""
        comp = self._current_composite.get_component(comp_id)
        item = self._items_by_comp_id.get(comp_id)
        if comp is None or item is None:
            return

        if field not in ("x", "y", "rotation", "scale", "layer"):
            return

        old = getattr(comp, field, None)
        if old is None:
            return

        new = int(value) if field == "layer" else float(value)

        if abs(float(old) - float(new)) < 1e-9:
            return

        cmd = SetPropertyCommand(
            comp, field, old, new,
            on_apply=lambda: self._apply_property(comp_id),
        )
        self._undo_stack.push(cmd)

    def _on_prop_layer_shift(
        self, comp_id: str, direction: int,
    ) -> None:
        """РЎРґРІРёРЅСѓС‚СЊ СЃР»РѕР№ РЅР° +1 / -1 вЂ” С‡РµСЂРµР· undo-РєРѕРјР°РЅРґСѓ."""
        comp = self._current_composite.get_component(comp_id)
        if comp is None:
            return

        old = int(getattr(comp, "layer", 0))
        new = old + int(direction)

        cmd = SetPropertyCommand(
            comp, "layer", old, new,
            on_apply=lambda: self._apply_property(comp_id),
        )
        self._undo_stack.push(cmd)

    def _on_prop_filled_changed(

        item.update()

        new = str(pattern or "")
        if old == new:

        def _apply():
            it = self._items_by_comp_id.get(comp_id)
            if it is not None:
                it.update()

        cmd = SetPropertyCommand(

    def _on_prop_locked_changed(

    def _apply_override(self, comp_id: str, name: str) -> None:
        """РџСЂРёРјРµРЅРёС‚СЊ override Рє item вЂ” РїРµСЂРµСЃРѕР±СЂР°С‚СЊ path + reflow."""

            item.prepareGeometryChange()
            item._rebuild_paths()
            item.update()

        from PySide6.QtCore import QTimer

    def _create_component_item(self, comp) -> "ComponentItem":
        """РЎРѕР·РґР°С‚СЊ ComponentItem РґР»СЏ РєРѕРјРїРѕРЅРµРЅС‚Р°, РґРѕР±Р°РІРёС‚СЊ РЅР° СЃС†РµРЅСѓ,
        РїРѕРґРєР»СЋС‡РёС‚СЊ РІСЃРµ СЃРёРіРЅР°Р»С‹. Р’РѕР·РІСЂР°С‰Р°РµС‚ item."""

    def _on_prop_param_override(
        self, comp_id: str, name: str, value: float,
    ) -> None:
        """РџРѕР»СЊР·РѕРІР°С‚РµР»СЊ РјРµРЅСЏРµС‚ РїР°СЂР°РјРµС‚СЂ sub-Р°СЃСЃРµС‚Р° вЂ” С‡РµСЂРµР· РєРѕРјР°РЅРґСѓ."""

            return

        # Р•СЃР»Рё Р·РЅР°С‡РµРЅРёРµ РЅРµ РёР·РјРµРЅРёР»РѕСЃСЊ вЂ” РЅРёС‡РµРіРѕ РЅРµ РґРµР»Р°РµРј
        if old is not None and abs(float(old) - float(value)) < 1e-9:
            return

        cmd = SetParamOverrideCommand(
            comp, name, old, value,
            on_apply=lambda: self._apply_override(comp_id, name),
        )
        self._undo_stack.push(cmd)

        # РЎРёРЅС…СЂРѕРЅРёР·Р°С†РёСЏ СЃРїРёРЅР±РѕРєСЃР° РІ РїР°РЅРµР»Рё (РµСЃР»Рё РјРµРЅСЏР»Рё С…РµРЅРґР»РѕРј)
        self._properties.sync_param_value(comp_id, name, value)

        # V11: СЃСЂР°Р·Сѓ РїРµСЂРµСЃС‡РёС‚Р°С‚СЊ РІРёРґРёРјРѕСЃС‚СЊ СЃР»РѕС‚РѕРІ
        from PySide6.QtCore import QTimer as _QT2
        _QT2.singleShot(0, self._apply_slot_visibility)

    def _get_selected_comps(self) -> list:

    def _on_copy(self) -> None:
        comps = self._get_selected_comps()
        if not comps:
            return
        # РљРѕРїРёСЂСѓРµРј С‡РµСЂРµР· dict вЂ” РѕС‚РґРµР»СЊРЅС‹Рµ РѕР±СЉРµРєС‚С‹
        self._clipboard_comps = [
            Component.from_dict(c.to_dict()) for c in comps
        ]
        n = len(self._clipboard_comps)
        self.statusBar().showMessage(
            f"РЎРєРѕРїРёСЂРѕРІР°РЅРѕ: {n} С€С‚.", 2000,
        )

    def _spawn_components(self, sources: list) -> list:
        """РЎРѕР·РґР°С‚СЊ РєРѕРїРёРё РєРѕРјРїРѕРЅРµРЅС‚РѕРІ СЃРѕ СЃРґРІРёРіРѕРј, С‡РµСЂРµР· Add РєРѕРјР°РЅРґС‹."""
        new_ids = []
        for src_comp in sources:
            new_comp = Component(
                asset_id=src_comp.asset_id,
                x=src_comp.x + 1.0,
                y=src_comp.y + 1.0,
                rotation=src_comp.rotation,
                scale=src_comp.scale,
                name=src_comp.name,
                layer=src_comp.layer,
                filled=src_comp.filled,
                param_overrides=dict(src_comp.param_overrides),
            )
            cmd = AddComponentCommand(
                composite=self._current_composite,
                comp=new_comp,
                on_create_item=self._create_component_item,
                on_remove_item=self._remove_component_item,
            )
            self._undo_stack.push(cmd)
            new_ids.append(new_comp.id)
        return new_ids

    def _on_paste(self) -> None:
        if not self._clipboard_comps:
            self.statusBar().showMessage("Р‘СѓС„РµСЂ РїСѓСЃС‚", 2000)
            return

        new_ids = self._spawn_components(self._clipboard_comps)

        from PySide6.QtCore import QTimer
        QTimer.singleShot(0, lambda: self._select_components(new_ids))

        self.statusBar().showMessage(
            f"Р’СЃС‚Р°РІР»РµРЅРѕ: {len(new_ids)} С€С‚.", 2000,
        )

    def _on_duplicate(self) -> None:
        comps = self._get_selected_comps()
        if not comps:
            return

        new_ids = self._spawn_components(comps)

        from PySide6.QtCore import QTimer
        QTimer.singleShot(0, lambda: self._select_components(new_ids))

        self.statusBar().showMessage(
            f"Р”СѓР±Р»РёСЂРѕРІР°РЅРѕ: {len(new_ids)} С€С‚.", 2000,
        )

    def _select_components(self, comp_ids: list) -> None:
        self._scene.clearSelection()
        for cid in comp_ids:
            item = self._items_by_comp_id.get(cid)
            if item is not None:
                item.setSelected(True)

        """РЈРґР°Р»РёС‚СЊ РєРѕРјРїРѕРЅРµРЅС‚ С‡РµСЂРµР· undo-РєРѕРјР°РЅРґСѓ."""

    def _reflow_children(
        self, parent_id: str, visited: set | None = None,
    ) -> None:

            if child_id in visited:
                continue
            child_comp = self._current_composite.get_component(child_id)
            if child_comp is None:
                continue
            if child_comp.attach_to != parent_id:
                continue
            if not child_comp.attach_anchor or not child_comp.parent_anchor:
                continue

            # V9d-4: РµСЃР»Рё parent_anchor вЂ” СЃР»РѕС‚ (slot_*),

            if p_local is None or c_local is None:

                QPointF(p_local[0], p_local[1])

            new_y = parent_scene.y() - c_local[1]

            child_comp.x = new_x
            child_comp.y = new_y
            child_item.setPos(new_x, new_y)
            child_item.update()
